   106→   106→function deriveSkinColor(tag: string): string {
   107→   107→  const palette = ['#10b981', '#a855f7', '#eab308', '#ef4444', '#06b6d4', '#f97316', '#ec4899', '#8b5cf6'];
   108→   108→  let hash = 0;
   109→   109→  for (let i = 0; i < tag.length; i++) hash = tag.charCodeAt(i) + ((hash << 5) - hash);
   110→   110→  return palette[Math.abs(hash) % palette.length];
   111→   111→}
