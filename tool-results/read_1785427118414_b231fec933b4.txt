   342→   342→// ---------------------------------------------------------------------------
   343→   343→// generateCustomSegments — exact replica of original helper.
   344→   344→// ---------------------------------------------------------------------------
   345→   345→function generateCustomSegments(
   346→   346→  colors: string[],
   347→   347→  shapeStyle: BodyStyle,
   348→   348→  taperStyle: TaperStyle,
   349→   349→  glowEnabled: boolean,
   350→   350→): CustomSegment[] {
   351→   351→  if (colors.length === 0) return [];
   352→   352→  const result: CustomSegment[] = [];
   353→   353→  const totalNodes = 16;
   354→   354→
   355→   355→  for (let i = 0; i < totalNodes; i++) {
   356→   356→    const color = colors[i % colors.length];
   357→   357→
   358→   358→    let shape: SegShape = 'circle';
   359→   359→    if (shapeStyle === 'smooth') {
   360→   360→      shape = 'circle';
   361→   361→    } else if (shapeStyle === 'dragon') {
   362→   362→      shape = i === 0 ? 'circle' : i % 2 === 1 ? 'spike' : 'circle';
   363→   363→    } else if (shapeStyle === 'armored') {
   364→   364→      shape = i === 0 ? 'circle' : i % 2 === 1 ? 'square' : 'circle';
   365→   365→    } else if (shapeStyle === 'crystal') {
   366→   366→      shape = i === 0 ? 'circle' : i % 2 === 1 ? 'diamond' : 'circle';
   367→   367→    } else if (shapeStyle === 'obsidian') {
   368→   368→      shape = 'spike';
   369→   369→    } else if (shapeStyle === 'basilisk') {
   370→   370→      shape = 'diamond';
   371→   371→    }
   372→   372→
   373→   373→    let sizeScale = 1.0;
   374→   374→    if (taperStyle === 'uniform') {
   375→   375→      sizeScale = i === 0 ? 1.3 : 1.0;
   376→   376→    } else if (taperStyle === 'natural') {
   377→   377→      sizeScale = i === 0 ? 1.35 : Math.max(0.65, 1.25 - (i / totalNodes) * 0.55);
   378→   378→    } else if (taperStyle === 'wave') {
   379→   379→      sizeScale = i === 0 ? 1.3 : 1.0 + Math.sin(i * 0.95) * 0.22;
   380→   380→    } else if (taperStyle === 'heavy') {
   381→   381→      sizeScale = i === 0 ? 1.6 : Math.max(0.55, 1.35 - (i / totalNodes) * 0.8);
   382→   382→    }
   383→   383→
   384→   384→    result.push({
   385→   385→      color,
   386→   386→      shape,
   387→   387→      glow: glowEnabled,
   388→   388→      sizeScale: Number(sizeScale.toFixed(2)),
   389→   389→    });
   390→   390→  }
   391→   391→
   392→   392→  return result;
   393→   393→}

   443→   443→function SkinsCanvasPreview({
   444→   444→  colors,
   445→   445→  pattern,
   446→   446→  shapeStyle = 'smooth',
   447→   447→  glow = false,
   448→   448→}: SkinsCanvasPreviewProps) {
   449→   449→  const canvasRef = useRef<HTMLCanvasElement>(null);
   450→   450→
   451→   451→  useEffect(() => {
   452→   452→    const canvas = canvasRef.current;
   453→   453→    if (!canvas) return;
   454→   454→    const ctx = canvas.getContext('2d');
   455→   455→    if (!ctx) return;
   456→   456→
   457→   457→    let animId = 0;
   458→   458→    let time = Math.random() * 100;
   459→   459→
   460→   460→    const numSegments = 10;
   461→   461→    const points: Array<{ x: number; y: number }> = [];
   462→   462→    for (let i = 0; i < numSegments; i++) points.push({ x: 0, y: 0 });
   463→   463→
   464→   464→    const render = () => {
   465→   465→      ctx.clearRect(0, 0, canvas.width, canvas.height);
   466→   466→      time += 0.07;
   467→   467→
   468→   468→      // Grid background
   469→   469→      ctx.strokeStyle = 'rgba(99, 102, 241, 0.05)';
   470→   470→      ctx.lineWidth = 1;
   471→   471→      for (let x = 15; x < canvas.width; x += 30) {
   472→   472→        ctx.beginPath();
   473→   473→        ctx.moveTo(x, 0);
   474→   474→        ctx.lineTo(x, canvas.height);
   475→   475→        ctx.stroke();
   476→   476→      }
   477→   477→
   478→   478→      for (let i = 0; i < numSegments; i++) {
   479→   479→        const x = canvas.width - 24 - (i * (canvas.width - 48)) / (numSegments - 1);
   480→   480→        const wiggleVal = Math.sin(time - i * 0.42) * 9;
   481→   481→        const y = canvas.height / 2 + wiggleVal;
   482→   482→        points[i] = { x, y };
   483→   483→      }
   484→   484→
   485→   485→      // Draw trailing segment shadows & nodes from tail to head
   486→   486→      for (let i = points.length - 1; i >= 1; i--) {
   487→   487→        const pt = points[i];
   488→   488→        const nextPt = points[i - 1] || pt;
   489→   489→        const segAngle = Math.atan2(nextPt.y - pt.y, nextPt.x - pt.x);
   490→   490→
   491→   491→        let fillColor = colors[i % colors.length] || '#ffffff';
   492→   492→        let segmentGlow = glow;
   493→   493→
   494→   494→        if (pattern === 'rainbow') {
   495→   495→          const hue = (Date.now() * 0.06 + i * 36) % 360;
   496→   496→          fillColor = `hsl(${hue}, 85%, 55%)`;
   497→   497→          segmentGlow = true;
   498→   498→        } else if (pattern === 'neon') {
   499→   499→          fillColor = i % 2 === 0 ? '#06b6d4' : '#a855f7';
   500→   500→          segmentGlow = true;
   501→   501→        } else if (pattern === 'metallic') {
   502→   502→          fillColor = i % 2 === 0 ? '#cbd5e1' : '#475569';
   503→   503→        } else if (pattern === 'camo') {
   504→   504→          fillColor = i % 2 === 0 ? '#15803d' : '#854d0e';
   505→   505→        }
   506→   506→
   507→   507→        const sizeRatio = 1 - (i / points.length) * 0.42;
   508→   508→        const r = 6 * sizeRatio;
   509→   509→
   510→   510→        ctx.save();
   511→   511→        if (segmentGlow) {
   512→   512→          ctx.shadowBlur = 8;
   513→   513→          ctx.shadowColor = fillColor;
   514→   514→        }
   515→   515→        ctx.fillStyle = fillColor;
   516→   516→
   517→   517→        let segmentShape: SegShape = 'circle';
   518→   518→        if (shapeStyle === 'dragon') {
   519→   519→          segmentShape = i % 2 === 1 ? 'spike' : 'circle';
   520→   520→        } else if (shapeStyle === 'armored') {
   521→   521→          segmentShape = i % 2 === 1 ? 'square' : 'circle';
   522→   522→        } else if (shapeStyle === 'crystal') {
   523→   523→          segmentShape = i % 2 === 1 ? 'diamond' : 'circle';
   524→   524→        } else if (shapeStyle === 'obsidian') {
   525→   525→          segmentShape = 'spike';
   526→   526→        } else if (shapeStyle === 'basilisk') {
   527→   527→          segmentShape = 'diamond';
   528→   528→        }
   529→   529→
   530→   530→        ctx.beginPath();
   531→   531→        if (segmentShape === 'circle') {
   532→   532→          ctx.arc(pt.x, pt.y, r, 0, Math.PI * 2);
   533→   533→          ctx.fill();
   534→   534→        } else if (segmentShape === 'square') {
   535→   535→          ctx.fillRect(pt.x - r, pt.y - r, r * 2, r * 2);
   536→   536→        } else if (segmentShape === 'diamond') {
   537→   537→          ctx.moveTo(pt.x, pt.y - r);
   538→   538→          ctx.lineTo(pt.x + r, pt.y);
   539→   539→          ctx.lineTo(pt.x, pt.y + r);
   540→   540→          ctx.lineTo(pt.x - r, pt.y);
   541→   541→          ctx.closePath();
   542→   542→          ctx.fill();
   543→   543→        } else if (segmentShape === 'spike') {
   544→   544→          const perpAngle = segAngle + Math.PI / 2;
   545→   545→          ctx.moveTo(
   546→   546→            pt.x + Math.cos(segAngle) * r * 1.3,
   547→   547→            pt.y + Math.sin(segAngle) * r * 1.3,
   548→   548→          );
   549→   549→          ctx.lineTo(
   550→   550→            pt.x + Math.cos(perpAngle) * r * 0.8,
   551→   551→            pt.y + Math.sin(perpAngle) * r * 0.8,
   552→   552→          );
   553→   553→          ctx.lineTo(
   554→   554→            pt.x - Math.cos(segAngle) * r * 0.3,
   555→   555→            pt.y - Math.sin(segAngle) * r * 0.3,
   556→   556→          );
   557→   557→          ctx.lineTo(
   558→   558→            pt.x - Math.cos(perpAngle) * r * 0.8,
   559→   559→            pt.y - Math.sin(perpAngle) * r * 0.8,
   560→   560→          );
   561→   561→          ctx.closePath();
   562→   562→          ctx.fill();
   563→   563→        }
   564→   564→        ctx.restore();
   565→   565→      }
   566→   566→
   567→   567→      // Head
   568→   568→      const head = points[0];
   569→   569→      const prevPt = points[1] || head;
   570→   570→      const headAngle = Math.atan2(head.y - prevPt.y, head.x - prevPt.x);
   571→   571→
   572→   572→      let headColor = colors[0] || '#ffffff';
   573→   573→      if (pattern === 'rainbow') {
   574→   574→        const hue = (Date.now() * 0.06) % 360;
   575→   575→        headColor = `hsl(${hue}, 85%, 55%)`;
   576→   576→      } else if (pattern === 'neon') {
   577→   577→        headColor = '#06b6d4';
   578→   578→      }
   579→   579→
   580→   580→      ctx.save();
   581→   581→      ctx.fillStyle = headColor;
   582→   582→      if (glow) {
   583→   583→        ctx.shadowBlur = 10;
   584→   584→        ctx.shadowColor = headColor;
   585→   585→      }
   586→   586→      ctx.beginPath();
   587→   587→      ctx.arc(head.x, head.y, 8.5, 0, Math.PI * 2);
   588→   588→      ctx.fill();
   589→   589→      ctx.restore();
   590→   590→
   591→   591→      // Eyes
   592→   592→      const eyeL_Angle = headAngle + 0.52;
   593→   593→      const eyeR_Angle = headAngle - 0.52;
   594→   594→      const eyeL_Pos = {
   595→   595→        x: head.x + Math.cos(eyeL_Angle) * 3.8,
   596→   596→        y: head.y + Math.sin(eyeL_Angle) * 3.8,
   597→   597→      };
   598→   598→      const eyeR_Pos = {
   599→   599→        x: head.x + Math.cos(eyeR_Angle) * 3.8,
   600→   600→        y: head.y + Math.sin(eyeR_Angle) * 3.8,
   601→   601→      };
   602→   602→
   603→   603→      ctx.fillStyle = '#ffffff';
   604→   604→      ctx.beginPath();
   605→   605→      ctx.arc(eyeL_Pos.x, eyeL_Pos.y, 2, 0, Math.PI * 2);
   606→   606→      ctx.arc(eyeR_Pos.x, eyeR_Pos.y, 2, 0, Math.PI * 2);
   607→   607→      ctx.fill();
   608→   608→
   609→   609→      ctx.fillStyle = '#090d16';
   610→   610→      ctx.beginPath();
   611→   611→      ctx.arc(
   612→   612→        eyeL_Pos.x + Math.cos(headAngle) * 0.6,
   613→   613→        eyeL_Pos.y + Math.sin(headAngle) * 0.6,
   614→   614→        1,
   615→   615→        0,
   616→   616→        Math.PI * 2,
   617→   617→      );
   618→   618→      ctx.arc(
   619→   619→        eyeR_Pos.x + Math.cos(headAngle) * 0.6,
   620→   620→        eyeR_Pos.y + Math.sin(headAngle) * 0.6,
   621→   621→        1,
   622→   622→        0,
   623→   623→        Math.PI * 2,
   624→   624→      );
   625→   625→      ctx.fill();
   626→   626→
   627→   627→      animId = requestAnimationFrame(render);
   628→   628→    };
   629→   629→
   630→   630→    render();
   631→   631→    return () => cancelAnimationFrame(animId);
   632→   632→  }, [colors, pattern, shapeStyle, glow]);
   633→   633→
   634→   634→  return (
   635→   635→    <div className="relative overflow-hidden rounded-xl border border-slate-800/80 bg-slate-900/60 p-1 group">
   636→   636→      <canvas
   637→   637→        ref={canvasRef}
   638→   638→        width={180}
   639→   639→        height={80}
   640→   640→        className="block max-w-full h-[80px] w-full"
   641→   641→      />
   642→   642→    </div>
   643→   643→  );
   644→   644→}

   649→   649→interface TryOnPreviewProps {
   650→   650→  colors: string[];

   655→   655→
   656→   656→function TryOnPreview({
   657→   657→  colors,
   658→   658→  shapeStyle,
   659→   659→  taperStyle,
   660→   660→  glow,
   661→   661→}: TryOnPreviewProps) {
   662→   662→  const canvasRef = useRef<HTMLCanvasElement>(null);
   663→   663→  const mousePos = useRef({ x: 220, y: 100 });
   664→   664→  const isHovered = useRef(false);
   665→   665→
   666→   666→  useEffect(() => {
   667→   667→    const canvas = canvasRef.current;
   668→   668→    if (!canvas) return;
   669→   669→    const ctx = canvas.getContext('2d');
   670→   670→    if (!ctx) return;
   671→   671→

   675→   675→    const numPoints = 26;

   677→   677→    for (let i = 0; i < numPoints; i++) points.push({ x: 220, y: 100 });
   678→   678→
   679→   679→    const handleMouseMove = (e: MouseEvent) => {
   680→   680→      const rect = canvas.getBoundingClientRect();

   682→   682→        x: e.clientX - rect.left,

   691→   691→    };
   692→   692→

   694→   694→    canvas.addEventListener('mouseenter', handleMouseEnter);
   695→   695→    canvas.addEventListener('mouseleave', handleMouseLeave);
   696→   696→

   698→   698→    let headY = 100;
   699→   699→

   701→   701→      ctx.clearRect(0, 0, canvas.width, canvas.height);
   702→   702→      time += 0.045;
   703→   703→
   704→   704→      let targetX = canvas.width / 2 + Math.cos(time * 0.9) * 120;
   705→   705→      let targetY = canvas.height / 2 + Math.sin(time * 1.6) * 45;

   716→   716→      if (dist > 3) {
   717→   717→        const speed = isHovered.current ? 4.8 : 3.4;
   718→   718→        const angle = Math.atan2(dy, dx);
   719→   719→        headX += Math.cos(angle) * speed;
   720→   720→        headY += Math.sin(angle) * speed;
   721→   721→      }
   722→   722→
   723→   723→      points.unshift({ x: headX, y: headY });
   724→   724→      if (points.length > numPoints) points.pop();

   730→   730→        ctx.beginPath();
   731→   731→        ctx.moveTo(x, 0);

   733→   733→        ctx.stroke();
   734→   734→      }
   735→   735→      for (let y = 0; y < canvas.height; y += 20) {
   736→   736→        ctx.beginPath();
   737→   737→        ctx.moveTo(0, y);
   738→   738→        ctx.lineTo(canvas.width, y);
   739→   739→        ctx.stroke();
   740→   740→      }
   741→   741→
   742→   742→      // Mouse radar ring
   743→   743→      if (isHovered.current) {
   744→   744→        ctx.strokeStyle = 'rgba(99, 102, 241, 0.2)';
   745→   745→        ctx.beginPath();
   746→   746→        ctx.arc(mousePos.current.x, mousePos.current.y, 12, 0, Math.PI * 2);
   747→   747→        ctx.stroke();

   754→   754→        const segAngle = Math.atan2(pt.y - prevPt.y, pt.x - prevPt.x);
   755→   755→        const perpAngle = segAngle + Math.PI / 2;
   756→   756→
   757→   757→        const sizeRatio = 1 - (i / points.length) * 0.45;
   758→   758→        const color = colors[i % colors.length] || '#ffffff';
   759→   759→
   760→   760→        let shape: SegShape = 'circle';
   761→   761→        if (shapeStyle === 'smooth') {
   762→   762→          shape = 'circle';
   763→   763→        } else if (shapeStyle === 'dragon') {
   764→   764→          shape = i % 2 === 1 ? 'spike' : 'circle';
   765→   765→        } else if (shapeStyle === 'armored') {
   766→   766→          shape = i % 2 === 1 ? 'square' : 'circle';
   767→   767→        } else if (shapeStyle === 'crystal') {
   768→   768→          shape = i % 2 === 1 ? 'diamond' : 'circle';
   769→   769→        } else if (shapeStyle === 'obsidian') {
   770→   770→          shape = 'spike';
   771→   771→        } else if (shapeStyle === 'basilisk') {
   772→   772→          shape = 'diamond';
   773→   773→        }

   780→   780→        } else if (taperStyle === 'wave') {

   782→   782→        } else if (taperStyle === 'heavy') {

   822→   822→          ctx.lineTo(
   823→   823→            pt.x + Math.cos(perpAngle - Math.PI) * r * 0.95,

   825→   825→          );
   826→   826→          ctx.closePath();
   827→   827→          ctx.fill();
   828→   828→        }
   829→   829→        ctx.restore();
   830→   830→      }
   831→   831→
   832→   832→      // Head
   833→   833→      const head = points[0];
   834→   834→      const headColor = colors[0] || '#ffffff';

   836→   836→      if (glow) {
   837→   837→        ctx.shadowBlur = 15;
   838→   838→        ctx.shadowColor = headColor;
   839→   839→      }
   840→   840→      ctx.fillStyle = headColor;
   841→   841→      ctx.beginPath();
   842→   842→      ctx.arc(head.x, head.y, 12, 0, Math.PI * 2);

   844→   844→      ctx.restore();

   850→   850→        ctx.lineWidth = 2.5;
   851→   851→        ctx.lineCap = 'round';

   853→   853→        const headAngle = Math.atan2(head.y - nextPt.y, head.x - nextPt.x);

   880→   880→        y: head.y + Math.sin(headAngle + 0.45) * 6,
   881→   881→      };
   882→   882→      const eyeR = {
   883→   883→        x: head.x + Math.cos(headAngle - 0.45) * 6,
   884→   884→        y: head.y + Math.sin(headAngle - 0.45) * 6,
   885→   885→      };
   886→   886→
   887→   887→      ctx.fillStyle = '#ffffff';
   888→   888→      ctx.beginPath();
   889→   889→      ctx.arc(eyeL.x, eyeL.y, 2.8, 0, Math.PI * 2);
   890→   890→      ctx.arc(eyeR.x, eyeR.y, 2.8, 0, Math.PI * 2);
   891→   891→      ctx.fill();
   892→   892→
   893→   893→      ctx.fillStyle = '#090d16';
   894→   894→      ctx.beginPath();
   895→   895→      ctx.arc(
   896→   896→        eyeL.x + Math.cos(headAngle) * 0.8,
   897→   897→        eyeL.y + Math.sin(headAngle) * 0.8,
   898→   898→        1.4,
   899→   899→        0,
   900→   900→        Math.PI * 2,
   901→   901→      );
   902→   902→      ctx.arc(
   903→   903→        eyeR.x + Math.cos(headAngle) * 0.8,
   904→   904→        eyeR.y + Math.sin(headAngle) * 0.8,
   905→   905→        1.4,
   906→   906→        0,
   907→   907→        Math.PI * 2,
   908→   908→      );
   909→   909→      ctx.fill();
   910→   910→
   911→   911→      animId = requestAnimationFrame(render);
   912→   912→    };
   913→   913→
   914→   914→    render();
   915→   915→    return () => {
   916→   916→      cancelAnimationFrame(animId);
   917→   917→      canvas.removeEventListener('mousemove', handleMouseMove);
   918→   918→      canvas.removeEventListener('mouseenter', handleMouseEnter);
   919→   919→      canvas.removeEventListener('mouseleave', handleMouseLeave);
   920→   920→    };
   921→   921→  }, [colors, shapeStyle, taperStyle, glow]);
   922→   922→
