     1→'use client';
     2→
     3→import { useCallback, useEffect, useMemo, useState } from 'react';
     4→import { useAuth } from '@/components/providers/auth-provider';
     5→import { countryFlag, type InspectedPlayer } from '@/lib/game-config';
     6→import {
     7→  GlowBlob,
     8→  MicroLabel,
     9→  NotSignedIn,
    10→  PanelSkeleton,
    11→  notify,
    12→  type ToastFn,
    13→} from './_panel-primitives';
    14→import {
    15→  Shield,
    16→  Search,
    17→  Plus,
    18→  Trophy,
    19→  Coins,
    20→  Users,
    21→  MessageSquare,
    22→  Send,
    23→  Award,
    24→  Check,
    25→  X,
    26→  Loader2,
    27→} from 'lucide-react';
    28→
    29→interface ClanSystemProps {
    30→  onToast?: ToastFn;
    31→  onInspectPlayer?: (p: InspectedPlayer) => void;
    32→}
    33→
    34→type Tab = 'mine' | 'browse' | 'form';
    35→
    36→interface ClanInfo {
    37→  tag: string;
    38→  name: string;
    39→  emblem: string;
    40→  description: string;
    41→  level: number;
    42→  bankedChips: number;
    43→  memberCount: number;
    44→}
    45→
    46→interface ClanMember {
    47→  userTag: string;
    48→  name: string;
    49→  country: string;
    50→  level: number;
    51→  bankedChips: number;
    52→  clanRank: string | null;
    53→  avatar: string | null;
    54→}
    55→
    56→interface ChatMessage {
    57→  id: string;
    58→  senderTag: string;
    59→  senderName: string;
    60→  rank: string;
    61→  message: string;
    62→  createdAt: string;
    63→}
    64→
    65→const EMBLEM_OPTIONS = [
    66→  { value: '🐍', label: '🐍 Viper Snake' },
    67→  { value: '👑', label: '👑 Royal Crown' },
    68→  { value: '🥷', label: '🥷 Cyber Ninja' },
    69→  { value: '🔥', label: '🔥 Phoenix Fire' },
    70→  { value: '⚡', label: '⚡ Lightning Bolt' },
    71→  { value: '💎', label: '💎 Diamond Shield' },
    72→];
    73→
    74→export function ClanSystem({ onToast, onInspectPlayer }: ClanSystemProps) {
    75→  const { player, refresh } = useAuth();
    76→  const [tab, setTab] = useState<Tab>('mine');
    77→  const [search, setSearch] = useState('');
    78→  const [depositAmount, setDepositAmount] = useState('');
    79→  const [broadcast, setBroadcast] = useState('');
    80→  const [formState, setFormState] = useState({
    81→    name: '',
    82→    tag: '',
    83→    motto: '',
    84→    emblem: EMBLEM_OPTIONS[0].value,
    85→    description: '',
    86→  });
    87→  const [formBusy, setFormBusy] = useState(false);
    88→  const [clans, setClans] = useState<ClanInfo[]>([]);
    89→  const [clansLoading, setClansLoading] = useState(false);
    90→  const [members, setMembers] = useState<ClanMember[]>([]);
    91→  const [membersLoading, setMembersLoading] = useState(false);
    92→  const [chatMessages, setChatMessages] = useState<ChatMessage[]>([]);
    93→  const [chatLoading, setChatLoading] = useState(false);
    94→  const [actionBusy, setActionBusy] = useState('');
    95→
    96→  const playerClanTag = player?.clanTag || null;
    97→
    98→  // Fetch clan directory for Browse tab
    99→  const fetchClans = useCallback(async () => {
   100→    setClansLoading(true);
   101→    try {
   102→      const res = await fetch('/api/clans/list', { cache: 'no-store' });
   103→      const data = (await res.json().catch(() => ({}))) as { clans?: ClanInfo[] };
   104→      setClans(data.clans || []);
   105→    } catch {
   106→      setClans([]);
   107→    } finally {
   108→      setClansLoading(false);
   109→    }
   110→  }, []);
   111→
   112→  // Fetch members when player is in a clan
   113→  const fetchMembers = useCallback(async (tag: string) => {
   114→    setMembersLoading(true);
   115→    try {
   116→      const res = await fetch('/api/clans/list', { cache: 'no-store' });
   117→      const data = (await res.json().catch(() => ({}))) as { clans?: ClanInfo[] };
   118→      const clan = (data.clans || []).find((c) => c.tag === tag);
   119→      if (!clan) { setMembers([]); return; }
   120→      // Fetch all clan members via a leaderboard-like approach
   121→      // We'll use the chat members as a proxy, but better: fetch from player list
   122→      const res2 = await fetch(`/api/leaderboard?type=chips&limit=100`, { cache: 'no-store' });
   123→      const lbData = (await res2.json().catch(() => ({}))) as { entries?: LeaderboardEntry[] };
   124→      // Filter players with this clanTag (this is approximate — we show the clan info from the clan itself)
   125→      const clanMembers: ClanMember[] = (lbData.entries || [])
   126→        .filter((e) => e.userTag.includes(tag))
   127→        .slice(0, 30)
   128→        .map((e) => ({
   129→          userTag: e.userTag,
   130→          name: e.name,
   131→          country: e.country,
   132→          level: e.level,
   133→          bankedChips: e.bankedChips,
   134→          clanRank: null,
   135→          avatar: null,
   136→        }));
   137→      setMembers(clanMembers);
   138→    } catch {
   139→      setMembers([]);
   140→    } finally {
   141→      setMembersLoading(false);
   142→    }
   143→  }, []);
   144→
   145→  // Fetch chat messages for My Clan tab
   146→  const fetchChat = useCallback(async (tag: string) => {
   147→    setChatLoading(true);
   148→    try {
   149→      const res = await fetch(`/api/clans/chat?tag=${encodeURIComponent(tag)}`, { cache: 'no-store' });
   150→      const data = (await res.json().catch(() => ({}))) as { messages?: ChatMessage[] };
   151→      setChatMessages(data.messages || []);
   152→    } catch {
   153→      setChatMessages([]);
   154→    } finally {
   155→      setChatLoading(false);
   156→    }
   157→  }, []);
   158→
   159→  // Load data on mount and tab changes
   160→  useEffect(() => {
   161→    if (tab === 'browse') void fetchClans();
   162→  }, [tab, fetchClans]);
   163→
   164→  useEffect(() => {
   165→    if (tab === 'mine' && playerClanTag) {
   166→      void fetchChat(playerClanTag);
   167→    }
   168→  }, [tab, playerClanTag, fetchChat]);
   169→
   170→  // Also refresh when player data changes (e.g. after joining/leaving)
   171→  useEffect(() => {
   172→    if (playerClanTag && tab === 'mine') {
   173→      void fetchChat(playerClanTag);
   174→      void fetchMembers(playerClanTag);
   175→    }
   176→  }, [playerClanTag, tab, fetchChat, fetchMembers]);
   177→
   178→  // Player's clan info from the directory
   179→  const myClanInfo = clans.find((c) => c.tag === playerClanTag);
   180→
   181→  const filteredClans = useMemo(() => {
   182→    if (!search.trim()) return clans;
   183→    const q = search.toLowerCase();
   184→    return clans.filter((c) => c.name.toLowerCase().includes(q) || c.tag.toLowerCase().includes(q));
   185→  }, [clans, search]);
   186→
   187→  if (!player) return <NotSignedIn />;
   188→
   189→  async function handleJoinClan(tag: string, clanName: string) {
   190→    if (playerClanTag) {
   191→      notify('You are already in a clan! Leave your current clan first.', 'error', onToast);
   192→      return;
   193→    }
   194→    setActionBusy('join');
   195→    try {
   196→      const res = await fetch('/api/clans/join', {
   197→        method: 'POST',
   198→        headers: { 'Content-Type': 'application/json' },
   199→        body: JSON.stringify({ tag }),
   200→      });
   201→      const data = (await res.json().catch(() => ({}))) as { ok?: boolean; error?: string };
   202→      if (!res.ok) {
   203→        notify(data?.error || 'Failed to join clan.', 'error', onToast);
   204→        return;
   205→      }
   206→      notify(`Welcome to ${clanName} [${tag}]! 🛡️`, 'success', onToast);
   207→      await refresh();
   208→      void fetchClans();
   209→    } catch {
   210→      notify('Network error joining clan.', 'error', onToast);
   211→    } finally {
   212→      setActionBusy('');
   213→    }
   214→  }
   215→
   216→  async function handleLeaveClan() {
   217→    if (!playerClanTag) return;
   218→    setActionBusy('leave');
   219→    try {
   220→      const res = await fetch('/api/clans/leave', { method: 'POST' });
   221→      const data = (await res.json().catch(() => ({}))) as { ok?: boolean; error?: string };
   222→      if (!res.ok) {
   223→        notify(data?.error || 'Failed to leave clan.', 'error', onToast);
   224→        return;
   225→      }
   226→      const clanName = myClanInfo?.name || playerClanTag;
   227→      notify(`Left ${clanName} [${playerClanTag}].`, 'info', onToast);
   228→      await refresh();
   229→      void fetchClans();
   230→      setChatMessages([]);
   231→      setMembers([]);
   232→    } catch {
   233→      notify('Network error leaving clan.', 'error', onToast);
   234→    } finally {
   235→      setActionBusy('');
   236→    }
   237→  }
   238→
   239→  async function handleDeposit() {
   240→    const amt = parseInt(depositAmount, 10);
   241→    if (!amt || amt <= 0) {
   242→      notify('Enter a valid deposit amount.', 'error', onToast);
   243→      return;
   244→    }
   245→    if (amt > player.bankedChips) {
   246→      notify('Insufficient chips to deposit.', 'error', onToast);
   247→      return;
   248→    }
   249→    if (!playerClanTag) {
   250→      notify('You must be in a clan to deposit.', 'error', onToast);
   251→      return;
   252→    }
   253→    setActionBusy('deposit');
   254→    try {
   255→      const res = await fetch('/api/clans/deposit', {
   256→        method: 'POST',
   257→        headers: { 'Content-Type': 'application/json' },
   258→        body: JSON.stringify({ tag: playerClanTag, amount: amt }),
   259→      });
   260→      const data = (await res.json().catch(() => ({}))) as { ok?: boolean; error?: string; newTreasury?: number };
   261→      if (!res.ok) {
   262→        notify(data?.error || 'Failed to deposit.', 'error', onToast);
   263→        return;
   264→      }
   265→      notify(`Deposited ${amt.toLocaleString()}c to clan treasury!`, 'success', onToast);
   266→      setDepositAmount('');
   267→      await refresh();
   268→      void fetchClans();
   269→    } catch {
   270→      notify('Network error depositing chips.', 'error', onToast);
   271→    } finally {
   272→      setActionBusy('');
   273→    }
   274→  }
   275→
   276→  async function handleBroadcast() {
   277→    if (!broadcast.trim() || !playerClanTag) return;
   278→    setActionBusy('broadcast');
   279→    try {
   280→      const res = await fetch('/api/clans/chat', {
   281→        method: 'POST',
   282→        headers: { 'Content-Type': 'application/json' },
   283→        body: JSON.stringify({ tag: playerClanTag, message: broadcast.trim() }),
   284→      });
   285→      const data = (await res.json().catch(() => ({}))) as { ok?: boolean; error?: string };
   286→      if (!res.ok) {
   287→        notify(data?.error || 'Failed to send broadcast.', 'error', onToast);
   288→        return;
   289→      }
   290→      notify('Broadcast posted to clan chat! 📢', 'success', onToast);
   291→      setBroadcast('');
   292→      void fetchChat(playerClanTag);
   293→    } catch {
   294→      notify('Network error sending broadcast.', 'error', onToast);
   295→    } finally {
   296→      setActionBusy('');
   297→    }
   298→  }
   299→
   300→  async function handleFormSubmit() {
   301→    if (!formState.name.trim() || !formState.tag.trim()) {
   302→      notify('Syndicate Name and Clan Tag are required.', 'error', onToast);
   303→      return;
   304→    }
   305→    if (playerClanTag) {
   306→      notify('Leave your current clan before forming a new one.', 'error', onToast);
   307→      return;
   308→    }
   309→    setFormBusy(true);
   310→    try {
   311→      const res = await fetch('/api/clans/create', {
   312→        method: 'POST',
   313→        headers: { 'Content-Type': 'application/json' },
   314→        body: JSON.stringify({
   315→          tag: formState.tag,
   316→          name: formState.name,
   317→          emblem: formState.emblem,
   318→          description: formState.motto || formState.description,
   319→        }),
   320→      });
   321→      const data = (await res.json().catch(() => ({}))) as { ok?: boolean; error?: string; clanTag?: string };
   322→      if (!res.ok) {
   323→        notify(data?.error || 'Failed to create clan.', 'error', onToast);
   324→        return;
   325→      }
   326→      notify(`Syndicate "${formState.name}" [${data.clanTag}] formed! You are the Leader.`, 'success', onToast);
   327→      setFormState({ name: '', tag: '', motto: '', emblem: EMBLEM_OPTIONS[0].value, description: '' });
   328→      setTab('mine');
   329→      await refresh();
   330→      void fetchClans();
   331→    } catch {
   332→      notify('Network error creating clan.', 'error', onToast);
   333→    } finally {
   334→      setFormBusy(false);
   335→    }
   336→  }
   337→
   338→  function inspectMember(m: ClanMember) {
   339→    if (!onInspectPlayer) return;
   340→    onInspectPlayer({
   341→      name: m.name,
   342→      userTag: m.userTag,
   343→      country: m.country,
   344→      flag: countryFlag(m.country),
   345→      bankedChips: m.bankedChips,
   346→      level: m.level,
   347→      clanTag: playerClanTag || undefined,
   348→      clanName: myClanInfo?.name || undefined,
   349→    });
   350→  }
   351→
   352→  return (
   353→    <div className="relative rounded-2xl border border-slate-800/80 bg-slate-900/60 shadow-md p-5 sm:p-6 overflow-hidden">
   354→      <GlowBlob color="bg-indigo-500/10" className="-top-12 -right-12 w-56 h-56" />
   355→
   356→      {/* Header */}
   357→      <div className="relative flex flex-col sm:flex-row sm:items-center justify-between gap-3 mb-5 pb-5 border-b border-slate-800">
   358→        <div>
   359→          <h2 className="text-xl sm:text-2xl font-sans font-black text-white tracking-tight flex items-center gap-2.5">
   360→            <Shield className="w-5.5 h-5.5 text-indigo-400" />
   361→            Viper Clan &amp; Syndicate Guild HQ
   362→          </h2>
   363→          <p className="text-xs text-slate-400 mt-1 max-w-3xl">
   364→            Form or join a player syndicate, pool chips into the Clan Treasury, level up for
   365→            extraction perks, and dominate Clan Wars!
   366→          </p>
   367→        </div>
   368→      </div>
   369→
   370→      {/* Tabs */}
   371→      <div className="relative flex flex-wrap items-center gap-1.5 bg-slate-950 p-1 rounded-xl border border-slate-800/60 mb-5">
   372→        <ClanTabBtn active={tab === 'mine'} onClick={() => setTab('mine')} icon={Shield} label="My Clan" />
   373→        <ClanTabBtn active={tab === 'browse'} onClick={() => setTab('browse')} icon={Search} label="Browse Clans" />
   374→        <ClanTabBtn active={tab === 'form'} onClick={() => setTab('form')} icon={Plus} label="Form Syndicate" />
   375→      </div>
   376→
   377→      {/* MY CLAN TAB */}
   378→      {tab === 'mine' && (
   379→        <div>
   380→          {!playerClanTag ? (
   381→            <div className="p-8 rounded-2xl border border-slate-800 bg-slate-950/60 text-center max-w-md mx-auto">
   382→              <Shield className="w-12 h-12 text-slate-600 mx-auto mb-3" />
   383→              <h3 className="text-base font-bold text-white">You are not in a Viper Clan</h3>
   384→              <p className="text-xs text-slate-400 mt-2 mb-4">
   385→                Join an existing clan from the directory to participate in Clan Wars and earn
   386→                extraction perks, or form your own syndicate!
   387→              </p>
   388→              <div className="flex items-center justify-center gap-2">
   389→                <button
   390→                  type="button"
   391→                  onClick={() => setTab('browse')}
   392→                  className="px-3 py-2 rounded-lg bg-indigo-600 hover:bg-indigo-500 text-white text-xs font-bold transition"
   393→                >
   394→                  Browse Clans
   395→                </button>
   396→                <button
   397→                  type="button"
   398→                  onClick={() => setTab('form')}
   399→                  className="px-3 py-2 rounded-lg bg-slate-950 border border-slate-800 hover:border-indigo-500/40 text-slate-300 hover:text-white text-xs font-bold transition"
   400→                >
   401→                  Form Syndicate
   402→                </button>
   403→              </div>
   404→            </div>
   405→          ) : (
   406→            <div className="space-y-4">
   407→              {/* Header */}
   408→              <div className="flex items-start justify-between gap-3 flex-wrap p-4 rounded-2xl border border-slate-800 bg-slate-950/60">
   409→                <div className="flex items-center gap-3 min-w-0">
   410→                  <span className="text-3xl" aria-hidden>{myClanInfo?.emblem || '🐍'}</span>
   411→                  <div className="min-w-0">
   412→                    <h3 className="text-base font-bold text-white flex items-center gap-2 flex-wrap">
   413→                      {myClanInfo?.name || playerClanTag}
   414→                      <span className="text-[10px] font-mono text-indigo-300 bg-indigo-500/10 border border-indigo-500/30 px-2 py-0.5 rounded">[{playerClanTag}]</span>
   415→                      {player?.clanRank && (
   416→                        <span className="text-[9px] font-bold text-amber-300 bg-amber-500/10 border border-amber-500/30 px-2 py-0.5 rounded-full">
   417→                          <Trophy className="w-2.5 h-2.5 inline mr-0.5" /> {player.clanRank}
   418→                        </span>
   419→                      )}
   420→                    </h3>
   421→                    {myClanInfo?.description && (
   422→                      <p className="text-[11px] text-slate-400 italic mt-0.5">&quot;{myClanInfo.description}&quot;</p>
   423→                    )}
   424→                  </div>
   425→                </div>
   426→                <button
   427→                  type="button"
   428→                  onClick={() => void handleLeaveClan()}
   429→                  disabled={actionBusy === 'leave'}
   430→                  className="px-3 py-2 rounded-lg bg-slate-950 hover:bg-rose-950/40 text-slate-300 hover:text-rose-400 border border-slate-800 hover:border-rose-500/30 text-xs font-bold transition flex items-center gap-1.5 disabled:opacity-50"
   431→                >
   432→                  {actionBusy === 'leave' ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <X className="w-3.5 h-3.5" />}
   433→                  Leave Syndicate
   434→                </button>
   435→              </div>
   436→
   437→              {/* Stats */}
   438→              <div className="grid grid-cols-1 sm:grid-cols-3 gap-3 text-[11px]">
   439→                <div className="p-3 rounded-xl border border-slate-800 bg-slate-950/60">
   440→                  <MicroLabel>YOUR RANK</MicroLabel>
   441→                  <div className="text-white mt-0.5">{player?.clanRank || 'Viper'}</div>
   442→                </div>
   443→                <div className="p-3 rounded-xl border border-slate-800 bg-slate-950/60">
   444→                  <MicroLabel>MEMBERS</MicroLabel>
   445→                  <div className="text-white mt-0.5">{myClanInfo?.memberCount || '?'} / 30</div>
   446→                </div>
   447→                <div className="p-3 rounded-xl border border-slate-800 bg-slate-950/60">
   448→                  <MicroLabel>CLAN LEVEL</MicroLabel>
   449→                  <div className="text-white mt-0.5">Lvl {myClanInfo?.level || 1}</div>
   450→                </div>
   451→              </div>
   452→
   453→              {/* Treasury */}
   454→              <div className="p-4 rounded-2xl border border-slate-800 bg-slate-950/60">
   455→                <div className="flex items-center justify-between mb-2 flex-wrap gap-2">
   456→                  <h4 className="text-sm font-bold text-white flex items-center gap-2">
   457→                    <Coins className="w-4 h-4 text-emerald-400" /> Clan Treasury Bank
   458→                  </h4>
   459→                  <span className="text-base font-mono font-bold text-emerald-400">
   460→                    {(myClanInfo?.bankedChips || 0).toLocaleString()} c
   461→                  </span>
   462→                </div>
   463→                <div className="flex items-center gap-2">
   464→                  <input
   465→                    type="number"
   466→                    value={depositAmount}
   467→                    onChange={(e) => setDepositAmount(e.target.value)}
   468→                    placeholder="Amt (e.g. 100)"
   469→                    className="flex-1 bg-slate-950 border border-slate-800 rounded-lg px-3 py-2 text-xs text-white font-mono focus:outline-none focus:border-emerald-500/50"
   470→                  />
   471→                  <button
   472→                    type="button"
   473→                    onClick={() => void handleDeposit()}
   474→                    disabled={actionBusy === 'deposit'}
   475→                    className="px-3 py-2 rounded-lg bg-emerald-600 hover:bg-emerald-500 text-white text-xs font-bold transition flex items-center gap-1.5 disabled:opacity-50"
   476→                  >
   477→                    {actionBusy === 'deposit' ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Coins className="w-3.5 h-3.5" />}
   478→                    Deposit
   479→                  </button>
   480→                </div>
   481→              </div>
   482→
   483→              {/* Perks */}
   484→              <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   485→                {[
   486→                  { title: 'Self-Sponsored Arenas', desc: 'Host custom clan tournaments funded by Treasury' },
   487→                  { title: 'Clan Tag Emblem', desc: `Displays [${playerClanTag}] badge in match leaderboards` },
   488→                  { title: 'Syndicate Wars Access', desc: 'Qualified for weekly Clan vs Clan prize matches' },
   489→                ].map((p) => (
   490→                  <div key={p.title} className="p-3 rounded-xl border border-slate-800 bg-slate-950/60">
   491→                    <h5 className="text-xs font-bold text-white">{p.title}</h5>
   492→                    <p className="text-[11px] text-slate-400 mt-0.5">{p.desc}</p>
   493→                  </div>
   494→                ))}
   495→              </div>
   496→
   497→              {/* Members */}
   498→              <div>
   499→                <div className="flex items-center justify-between mb-2">
   500→                  <h4 className="text-sm font-bold text-white flex items-center gap-2">
   501→                    <Users className="w-4 h-4 text-indigo-400" /> Member Roster ({myClanInfo?.memberCount || 0})
   502→                  </h4>
   503→                  <span className="text-[10px] font-mono text-slate-500">Max Capacity: 30</span>
   504→                </div>
   505→                {membersLoading ? <PanelSkeleton count={3} height="h-12" /> : (
   506→                  <div className="rounded-2xl border border-slate-800/60 bg-slate-950/80 overflow-hidden">
   507→                    {members.length === 0 ? (
   508→                      <div className="p-6 text-center text-xs text-slate-500">
