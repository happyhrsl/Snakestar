   194→// ----------------------------------------------------------------------------
   195→// Spatial hash grid — slimmed-down client port of the server's grid.
   196→// Items are bucketed into square cells; queries return a deduplicated Map.
   197→// ----------------------------------------------------------------------------
   198→
   199→class SpatialHashGrid {
   200→  private readonly cellSize: number;
   201→  private readonly cells: Map<string, Map<string, GridItem>> = new Map();
   202→
   203→  constructor(cellSize = 120) {
   204→    this.cellSize = cellSize;
   205→  }
   206→
   207→  private key(cx: number, cy: number): string {
   208→    return cx + ':' + cy;
   209→  }
   210→
   211→  clear(): void {
   212→    this.cells.clear();
   213→  }
   214→
   215→  insert(item: GridItem): void {
   216→    const minCx = Math.floor((item.x - item.radius) / this.cellSize);
   217→    const maxCx = Math.floor((item.x + item.radius) / this.cellSize);
   218→    const minCy = Math.floor((item.y - item.radius) / this.cellSize);
   219→    const maxCy = Math.floor((item.y + item.radius) / this.cellSize);
   220→    for (let cx = minCx; cx <= maxCx; cx++) {
   221→      for (let cy = minCy; cy <= maxCy; cy++) {
   222→        const k = this.key(cx, cy);
   223→        let bucket = this.cells.get(k);
   224→        if (!bucket) {
   225→          bucket = new Map();
   226→          this.cells.set(k, bucket);
   227→        }
   228→        bucket.set(item.id, item);
   229→      }
   230→    }
   231→  }
   232→
   233→  queryRadius(x: number, y: number, r: number): Map<string, GridItem> {
   234→    const out = new Map<string, GridItem>();
   235→    const minCx = Math.floor((x - r) / this.cellSize);
   236→    const maxCx = Math.floor((x + r) / this.cellSize);
   237→    const minCy = Math.floor((y - r) / this.cellSize);
   238→    const maxCy = Math.floor((y + r) / this.cellSize);
   239→    for (let cx = minCx; cx <= maxCx; cx++) {
   240→      for (let cy = minCy; cy <= maxCy; cy++) {
   241→        const bucket = this.cells.get(this.key(cx, cy));
   242→        if (!bucket) continue;
   243→        for (const [id, item] of bucket) {
   244→          if (!out.has(id)) out.set(id, item);
   245→        }
   246→      }
   247→    }
   248→    return out;
   249→  }
   250→}

   320→function dist(ax: number, ay: number, bx: number, by: number): number {
   321→  return Math.hypot(ax - bx, ay - by);
   322→}

   324→/** Random point within a circle of maxR around center. */
   325→function randomPointInCircle(cx: number, cy: number, maxR: number): Vec2 {
   326→  const r = Math.sqrt(Math.random()) * maxR;
   327→  const theta = Math.random() * Math.PI * 2;
   328→  return {
   329→    x: cx + Math.cos(theta) * r,
   330→    y: cy + Math.sin(theta) * r,
   331→  };
   332→}
   333→
   334→function initialBody(headX: number, headY: number, angle: number, length: number, spacing: number): Vec2[] {
   335→  const pts: Vec2[] = [];
   336→  for (let i = 0; i < length; i++) {
   337→    pts.push({
   338→      x: headX - Math.cos(angle) * i * spacing,
   339→      y: headY - Math.sin(angle) * i * spacing,
   340→    });
   341→  }
   342→  return pts;
   343→}

   349→/** Create a Food object at a random position around the player. */
   350→function createFoodOrb(idPrefix: string, idCounter: { value: number }, cx: number, cy: number, farSpawn: boolean, foodOrbs: FoodOrbDef[]): Food {
   351→  const orb = randomFoodOrb(foodOrbs);
   352→  const radius = farSpawn ? FOOD_SPAWN_RADIUS_FAR : FOOD_SPAWN_RADIUS_NEAR;
   353→  const pos = randomPointInCircle(cx, cy, radius);
   354→  return {
   355→    id: `${idPrefix}-food-${idCounter.value++}`,
   356→    x: pos.x,
   357→    y: pos.y,
   358→    size: orb.radius,
   359→    value: orb.value,
   360→    orbSize: orb.size,
   361→    color: orb.color,
   362→    glowColor: orb.glowColor,
   363→  };
   364→}

   366→// ----------------------------------------------------------------------------
   367→// Death food drop — use engine calcDeathFood + distribute along body
   368→// ----------------------------------------------------------------------------
   369→
   370→function computeDeathFoodDrop(
   371→  totalScore: number,
   372→  bodyPoints: Vec2[],
   373→  idPrefix: string,
   374→  idCounter: { value: number },
   375→  foodOrbs: FoodOrbDef[],
   376→): Food[] {
   377→  const result: Food[] = [];
   378→  if (!bodyPoints || bodyPoints.length === 0 || totalScore <= 0) return result;
   379→
   380→  const [smallCount, mediumCount, largeCount] = calcDeathFood(totalScore, false);
   381→
   382→  // Build orb sequence from food definitions
   383→  const largeOrb = foodOrbs.find(o => o.size === 'large') ?? foodOrbs[0];
   384→  const mediumOrb = foodOrbs.find(o => o.size === 'medium') ?? foodOrbs[0];
   385→  const smallOrb = foodOrbs.find(o => o.size === 'small') ?? foodOrbs[0];
   386→
   387→  const orbSequence: FoodOrbDef[] = [];
   388→  for (let i = 0; i < largeCount; i++) orbSequence.push(largeOrb);
   389→  for (let i = 0; i < mediumCount; i++) orbSequence.push(mediumOrb);
   390→  for (let i = 0; i < smallCount; i++) orbSequence.push(smallOrb);
   391→
   392→  // Shuffle
   393→  for (let i = orbSequence.length - 1; i > 0; i--) {
   394→    const j = Math.floor(Math.random() * (i + 1));
   395→    [orbSequence[i], orbSequence[j]] = [orbSequence[j], orbSequence[i]];
   396→  }
   397→
   398→  // Distribute orbs evenly along the body
   399→  const totalOrbs = orbSequence.length;
   400→  if (totalOrbs === 0) return result;
   401→  const scatter = 15;
   402→  let orbIdx = 0;
   403→  for (const orb of orbSequence) {
   404→    const segIdx = Math.min(bodyPoints.length - 1, Math.floor((orbIdx / totalOrbs) * bodyPoints.length));
   405→    const pt = bodyPoints[segIdx];
   406→    result.push({
   407→      id: `${idPrefix}-death-${idCounter.value++}`,
   408→      x: pt.x + (Math.random() - 0.5) * scatter,
   409→      y: pt.y + (Math.random() - 0.5) * scatter,
   410→      size: orb.radius,
   411→      value: orb.value,
   412→      orbSize: orb.size,
   413→      color: orb.color,
   414→      glowColor: orb.glowColor,
   415→    });
   416→    orbIdx++;
   417→  }
   418→
   419→  return result;
   420→}

   727→  /** Create 1000 virtual bot definitions with cheap positions spread around (cx, cy). */
   728→  private createVirtualBotPool(cx: number, cy: number): void {
   729→    this.virtualBots = [];
   730→    for (let i = 0; i < VIRTUAL_BOT_COUNT; i++) {
   731→      const baseName = BOT_NAMES[i % BOT_NAMES.length];
   732→      const name = i < BOT_NAMES.length ? baseName : `${baseName}-${Math.floor(i / BOT_NAMES.length) + 1}`;
   733→      const skin = BOT_SKINS[i % BOT_SKINS.length];
   734→      const personality = PERSONALITIES[i % PERSONALITIES.length];
   735→      const botId = `bot-${this.arena.id}-${i}`;
   736→      const initialScore = Math.floor(Math.random() * 80);
   737→
   738→      // Spread bots in a large ring. Use a mix of near and far positions.
   739→      const spawn = randomPointInCircle(cx, cy, VIRTUAL_WORLD_RADIUS);
   740→      const botAngle = Math.random() * Math.PI * 2;
   741→
   742→      this.virtualBots.push({
   743→        idx: i,
   744→        id: botId,
   745→        botId,
   746→        name,
   747→        personality,

   749→        secondaryColor: skin.secondaryColor,
   750→        initialScore,
   751→        x: spawn.x,
   752→        y: spawn.y,
   753→        angle: botAngle,
   754→        score: initialScore,
   755→        isActive: false,
   756→      });
   757→    }
   758→  }

   760→  /** Cheaply update all virtual bot positions (straight-line wander). Called every tick but is very lightweight. */
   761→  private updateVirtualBotPositions(): void {
   762→    for (let i = 0; i < this.virtualBots.length; i++) {
   763→      const vb = this.virtualBots[i];
   764→      if (vb.isActive) continue; // active bots are moved by full physics
   765→      // Random angle tweak occasionally (use simple condition instead of Math.random for speed)
   766→      if (((this.tick + i) * 7919) % 1000 < 20) { // deterministic pseudo-random
   767→        vb.angle += (((this.tick + i) * 3571) % 1000 - 500) * 0.0016;
   768→      }
   769→      vb.x += Math.cos(vb.angle) * VIRTUAL_BOT_SPEED;
   770→      vb.y += Math.sin(vb.angle) * VIRTUAL_BOT_SPEED;
   771→    }
   772→  }

   774→  /** Activate virtual bots near the player, deactivate those far away. */
   775→  private activateNearbyBots(): void {
   776→    const p = this.player;
   777→    if (!p || p.points.length === 0) return;
   778→    const px = p.points[0].x;
   779→    const py = p.points[0].y;
   780→
   781→    // 1) Deactivate active bots that moved too far from the player.
   782→    const toDeactivate: string[] = [];
   783→    for (const [id, bot] of this.bots) {
   784→      if (bot.points.length === 0) continue;
   785→      const head = bot.points[0];
   786→      const d = dist(head.x, head.y, px, py);
   787→      if (d > DEACTIVATION_RADIUS) {
   788→        // Save state back to virtual bot
   789→        const vb = this.virtualBots[bot.virtualIdx];
   790→        if (vb) {
   791→          vb.x = head.x;
   792→          vb.y = head.y;
   793→          vb.angle = bot.angle;
   794→          vb.score = bot.score;
   795→          vb.isActive = false;
   796→        }
   797→        toDeactivate.push(id);
   798→      }
   799→    }
   800→    for (const id of toDeactivate) {
   801→      this.bots.delete(id);
   802→    }
   803→
   804→    // 2) Activate virtual bots within range (up to MAX_ACTIVE_BOTS).
   805→    // Only scan virtual pool every 10 ticks to reduce overhead.
   806→    if (this.tick % 10 !== 0) return;
   807→    if (this.bots.size >= MAX_ACTIVE_BOTS) return;
   808→
   809→    // Quick activation: scan virtual bots, find inactive ones within range.
   810→    // No sorting needed — just activate up to the limit.
   811→    for (let i = 0; i < this.virtualBots.length && this.bots.size < MAX_ACTIVE_BOTS; i++) {
   812→      const vb = this.virtualBots[i];
   813→      if (vb.isActive) continue;
   814→      const d = dist(vb.x, vb.y, px, py);
   815→      if (d < ACTIVATION_RADIUS) {
   816→        const bot = this.createActiveBotFromVirtual(vb, px, py);
   817→        this.bots.set(bot.id, bot);
   818→        vb.isActive = true;
   819→      }
   820→    }
   821→  }

   912→  private onTouchStart(e: TouchEvent): void {
   913→    if (this.joystick) return;
   914→    const t = this.findJoystickTouch(e.touches);
   915→    if (!t) return;
   916→    e.preventDefault();
   917→    const rect = this.canvas.getBoundingClientRect();
   918→    this.joystick = {
   919→      active: true,
   920→      pointerId: t.identifier,
   921→      originX: t.clientX - rect.left,
   922→      originY: t.clientY - rect.top,
   923→      curX: t.clientX - rect.left,
   924→      curY: t.clientY - rect.top,
   925→    };
   926→  }
   927→
   928→  private onTouchMove(e: TouchEvent): void {
   929→    const js = this.joystick;
   930→    if (!js) return;
   931→    let t: Touch | null = null;
   932→    for (let i = 0; i < e.touches.length; i++) {
   933→      if (e.touches[i].identifier === js.pointerId) {
   934→        t = e.touches[i];
   935→        break;
   936→      }
   937→    }
   938→    if (!t) return;
   939→    e.preventDefault();
   940→    const rect = this.canvas.getBoundingClientRect();
   941→    js.curX = t.clientX - rect.left;
   942→    js.curY = t.clientY - rect.top;
   943→    const dx = js.curX - js.originX;
   944→    const dy = js.curY - js.originY;
   945→    const d = Math.hypot(dx, dy);
   946→    const mag = Math.min(1, d / JOYSTICK_MAX_RADIUS_PX);
   947→    if (mag > JOYSTICK_DEADZONE) {
   948→      this.touchAngle = Math.atan2(dy, dx);
   949→      this.touchBoost = mag > JOYSTICK_BOOST_MAGNITUDE;
   950→    } else {
   951→      this.touchAngle = null;
   952→      this.touchBoost = false;
   953→    }
   954→  }
   955→
   956→  private onTouchEnd(e: TouchEvent): void {
   957→    const js = this.joystick;
   958→    if (!js) return;
   959→    let stillActive = false;
   960→    for (let i = 0; i < e.touches.length; i++) {
   961→      if (e.touches[i].identifier === js.pointerId) {
   962→        stillActive = true;
   963→        break;
   964→      }
   965→    }
   966→    if (!stillActive) {
   967→      this.joystick = null;
   968→      this.touchAngle = null;
   969→      this.touchBoost = false;
   970→    }
   971→  }
   972→
   973→  private findJoystickTouch(touches: TouchList): Touch | null {
   974→    const rect = this.canvas.getBoundingClientRect();
   975→    for (let i = 0; i < touches.length; i++) {
   976→      const t = touches[i];
   977→      const x = t.clientX - rect.left;
   978→      const y = t.clientY - rect.top;
   979→      if (x < rect.width / 2 && y > rect.height / 2) {
   980→        return t;
   981→      }
   982→    }
   983→    return null;
   984→  }

  1089→  private frame = (now: number): void => {
  1090→    if (this.stopped) return;
  1091→    this.rafId = requestAnimationFrame(this.frame);
  1092→    const dt = now - this.lastFrameTime;
  1093→    this.lastFrameTime = now;
  1094→
  1095→    this.updateFps(now);
  1096→    this.updateParticles(dt);
  1097→
  1098→    // Physics ticks at ~TICK_MS (30 Hz) cadence.
  1099→    if (this.state === 'playing' || this.isPostDeathRecording) {
  1100→      this.accumulator += dt;
  1101→      let safety = 4;
  1102→      while (this.accumulator >= TICK_MS && safety > 0) {
  1103→        this.tickPhysics(now);
  1104→        this.accumulator -= TICK_MS;
  1105→        safety--;
  1106→      }
  1107→      if (safety === 0) this.accumulator = 0;
  1108→    }
  1109→
  1110→    // Post-death recording: count down ticks
  1111→    if (this.isPostDeathRecording) {
  1112→      this.postDeathTicksRemaining--;
  1113→      if (this.postDeathTicksRemaining <= 0) {
  1114→        this.finishPostDeathRecording();
  1115→      }
  1116→    }
  1117→
  1118→    this.render(now);
  1119→    this.updateHUD();
  1120→  };

  1122→  private accumulator: number = 0;
  1123→
  1124→  // --------------------------------------------------------------------------
  1125→  // Physics tick (one server-tick equivalent)
  1126→  // --------------------------------------------------------------------------
  1127→
  1128→  private tickPhysics(now: number): void {
  1129→    this.tick++;
  1130→    const p = this.player;
  1131→    if (!p) return;
  1132→
  1133→    // During post-death recording, run simplified bot-only physics
  1134→    if (this.isPostDeathRecording) {
  1135→      this.tickPostDeathPhysics(now);
  1136→      this.captureReplaySnapshot();
  1137→      return;
  1138→    }
  1139→
  1140→    if (p.isDead) return;
  1141→
  1142→    // 1) Compute player desired angle + boost.
  1143→    const { angle, boost } = this.computePlayerInput();
  1144→    if (angle !== null) p.desiredAngle = angle;
  1145→    p.wantsBoost = boost;
  1146→    p.isBoosting = boost && p.points.length > this.cfg.boostMinLength && p.score > 0;
  1147→    // Boost activation sound
  1148→    if (p.isBoosting && !this.wasPlayerBoosting) {
  1149→      playBoost();
  1150→    }
  1151→    this.wasPlayerBoosting = p.isBoosting;
  1152→
  1153→    // Extraction progress (3-second hold).
  1154→    if (p.isExtracting) {
  1155→      p.extractionProgress += TICK_MS;
  1156→      if (p.extractionProgress >= this.cfg.extractionDurationMs) {
  1157→        this.finishExtract();
  1158→        return;
  1159→      }
  1160→    }
  1161→
  1162→    // 2) Move player.
  1163→    this.tickSnakeMovement(p, p.desiredAngle, p.wantsBoost);
  1164→
  1165→    // 3) Update virtual bot positions cheaply + manage activation/deactivation.
  1166→    this.updateVirtualBotPositions();
  1167→    this.activateNearbyBots();
  1168→
  1169→    // 4) Bots: think + move (only active bots).
  1170→    for (const bot of this.bots.values()) {
  1171→      this.tickBot(bot, now);
  1172→    }
  1173→
  1174→    // Process boost food drops (from tail shedding)
  1175→    if (this.boostDropQueue.length > 0) {
  1176→      for (const drop of this.boostDropQueue) {
  1177→        this.foods.push({
  1178→          id: `food-${this.arena.id}-${this.idCounterObj.value++}`,
  1179→          x: drop.x,
  1180→          y: drop.y,
  1181→          size: 3,
  1182→          value: 1,
  1183→          isStarChip: false,
  1184→          color: '#34d399',
  1185→          glowColor: '#10b981',
  1186→          orbSize: 'small',
  1187→        });
  1188→      }
  1189→      this.boostDropQueue.length = 0;
  1190→    }
  1191→
  1192→    // 4) Build spatial grid for collision + food queries.
  1193→    this.grid.clear();
  1194→    this.insertSnakeIntoGrid(p);
  1195→    for (const bot of this.bots.values()) {
  1196→      if (!bot.isDead) this.insertSnakeIntoGrid(bot);
  1197→    }
  1198→    for (const food of this.foods) {
  1199→      this.grid.insert({
  1200→        id: food.id,
  1201→        kind: 'food',
  1202→        x: food.x,
  1203→        y: food.y,
  1204→        radius: food.size,
  1205→        value: food.value,
  1206→        foodRef: food,
  1207→      });
  1208→    }
  1209→
  1210→    // 5) Eat food (player + bots).
  1211→    this.eatFood();
  1212→
  1213→    // 6) Collision detection (body collision + head-on collision). NO wall death.
  1214→    const deaths = this.detectCollisions(now);
  1215→
  1216→    // 7) Apply deaths + drop food.
  1217→    let playerDied = false;
  1218→    const newDropFoods: Food[] = [];
  1219→    for (const d of deaths) {
  1220→      if (d.deadId === p.id) {
  1221→        playerDied = true;
  1222→        // Drop player's food
  1223→        const playerTotal = this.cfg.initialSpawnScore + p.score;
  1224→        newDropFoods.push(...computeDeathFoodDrop(playerTotal, p.points, this.arena.id, this.idCounterObj, getFoodOrbs(this.cfg)));
  1225→        continue;
  1226→      }
  1227→      const bot = this.bots.get(d.deadId);
  1228→      if (bot) {
  1229→        // Drop bot's food
  1230→        const botTotal = this.cfg.initialSpawnScore + bot.score;
  1231→        newDropFoods.push(...computeDeathFoodDrop(botTotal, bot.points, this.arena.id, this.idCounterObj, getFoodOrbs(this.cfg)));
  1232→        bot.isDead = true;
  1233→        // Credit kill to the killer if it's the player.
  1234→        if (d.killerId === p.id) {
  1235→          p.kills++;
  1236→          playKill(); // Satisfying kill sound
  1237→        }
  1238→      }
  1239→    }
  1240→
  1241→    // Add dropped food to the world.
  1242→    if (newDropFoods.length > 0) {
  1243→      this.foods.push(...newDropFoods);
  1244→    }
  1245→
  1246→    // 8) Remove dead bots and recycle their virtual bots.
  1247→    if (deaths.length > 0) {
  1248→      const pHead = p.points[0];
  1249→      const toRemove: string[] = [];
  1250→      for (const [id, bot] of this.bots) {
  1251→        if (bot.isDead) {
  1252→          // Recycle the virtual bot to a new random position
  1253→          const vb = this.virtualBots[bot.virtualIdx];
  1254→          if (vb) {
  1255→            this.respawnVirtualBot(vb, pHead.x, pHead.y);
  1256→          }
  1257→          toRemove.push(id);
  1258→        }
  1259→      }
  1260→      for (const id of toRemove) {
  1261→        this.bots.delete(id);
  1262→      }
  1263→    }
  1264→
  1265→    // 9) Player death → enter post-death recording (15s of continued simulation).
  1266→    if (playerDied && !this.isPostDeathRecording) {
  1267→      p.isDead = true;
  1268→      playDeath(); // Dramatic crash sound on death
  1269→      this.enterPostDeathRecording();
  1270→      // Don't return — continue to replenish food, expire chat, update camera
  1271→    }
  1272→
  1273→    // 10) Replenish food toward target (spawn around player).
  1274→    this.replenishFood();
  1275→
  1276→    // 11) Expire chat bubbles.
  1277→    this.expireChat(now);
  1278→
  1279→    // 12) Update camera target.
  1280→    this.updateCamera();
  1281→
  1282→    // 13) Capture replay snapshot.
  1283→    this.captureReplaySnapshot();
  1284→  }

  1290→  private tickSnakeMovement(snake: SnakeBase, desiredAngle: number, wantsBoost: boolean): void {
  1291→    if (snake.points.length === 0 || snake.isDead) return;
  1292→
  1293→    // Turn rate from engine.
  1294→    const turnRate = calcTurnRate(snake.score, this.cfg);
  1295→    snake.angle = engineTurnToward(snake.angle, desiredAngle, turnRate);
  1296→
  1297→    // Speed from engine.
  1298→    const isBoosting = wantsBoost && snake.points.length > this.cfg.boostMinLength && snake.score > 0;
  1299→    snake.isBoosting = isBoosting;
  1300→    const speed = calcSpeed(isBoosting, snake.isExtracting, this.cfg);
  1301→
  1302→    if (isBoosting) {
  1303→      snake.boostFrameCounter++;
  1304→      if (snake.boostFrameCounter >= this.cfg.boostDropInterval) {
  1305→        snake.boostFrameCounter = 0;
  1306→        if (snake.points.length > this.cfg.boostMinLength + 1 && snake.score > 1) {
  1307→          const tail = snake.points[snake.points.length - 1];
  1308→          this.boostDropQueue.push({ x: tail.x, y: tail.y });
  1309→          snake.points.pop();
  1310→          snake.score = Math.max(0, snake.score - 1);
  1311→        }
  1312→      }
  1313→    }
  1314→
  1315→    // Move head using engine.
  1316→    const head = snake.points[0];
  1317→    const newHead = moveHead(head, snake.angle, speed);
  1318→    snake.points.unshift(newHead);
  1319→
  1320→    // Grow / shrink body to target length from engine.
  1321→    const targetLen = Math.max(
  1322→      this.cfg.boostMinLength + 1,
  1323→      calcBodyLength(snake.score, this.cfg),
  1324→    );
  1325→    while (snake.points.length > targetLen) snake.points.pop();
  1326→
  1327→    // Update visual + collision radii from engine.
  1328→    snake.size = calcVisualRadius(snake.score, this.cfg);
  1329→    snake.collisionRadius = calcCollisionRadius(snake.score, this.cfg);
  1330→  }

  1332→  // --------------------------------------------------------------------------
  1333→  // Bot AI — no self-destruct, seek food, predictive evasion
  1334→  // --------------------------------------------------------------------------
  1335→
  1336→  private tickBot(bot: BotSession, now: number): void {
  1337→    if (bot.points.length === 0 || bot.isDead) return;
  1338→
  1339→    if (now >= bot.nextThinkAt) {
  1340→      bot.nextThinkAt = now + BOT_THINK_INTERVAL_MS + Math.floor(Math.random() * BOT_THINK_JITTER_MS);
  1341→      const head = bot.points[0];
  1342→
  1343→      // --- Find nearest food ---
  1344→      let bestFood: GridItem | null = null;
  1345→      let bestFoodDist = Infinity;
  1346→      const foodQuery = this.grid.queryRadius(head.x, head.y, this.cfg.botFoodScanRadius);
  1347→      for (const item of foodQuery.values()) {
  1348→        if (item.kind !== 'food') continue;
  1349→        if ((item.value ?? 0) <= 0) continue;
  1350→        const d = dist(head.x, head.y, item.x, item.y);
  1351→        if (d < bestFoodDist) {
  1352→          bestFoodDist = d;
  1353→          bestFood = item;
  1354→        }
  1355→      }
  1356→
  1357→      // --- Find nearest threat (foreign body segment) ---
  1358→      let threatX = 0;
  1359→      let threatY = 0;
  1360→      let threatDist = Infinity;
  1361→      const threatQuery = this.grid.queryRadius(head.x, head.y, BOT_THREAT_SCAN_RADIUS);
  1362→      for (const item of threatQuery.values()) {
  1363→        if (item.kind !== 'segment') continue;
  1364→        if (item.snakeId === bot.id) continue;
  1365→        if (item.segIdx === 0) continue;
  1366→        const d = dist(head.x, head.y, item.x, item.y);
  1367→        if (d < threatDist) {
  1368→          threatDist = d;
  1369→          threatX = item.x;
  1370→          threatY = item.y;
  1371→        }
  1372→      }
  1373→
  1374→      let desired: number;
  1375→      const shouldFlee = threatDist < 150; // ALL bots dodge body segments when close
  1376→
  1377→      // --- Predictive evasion against the human player ---
  1378→      const p = this.player;
  1379→      if (p && !p.isDead && p.points.length > 0 && now >= p.spawnProtectedUntil) {
  1380→        const playerHead = p.points[0];
  1381→        const playerDist = dist(head.x, head.y, playerHead.x, playerHead.y);
  1382→        if (playerDist < this.cfg.botEvadeRadius * 2) {
  1383→          // Predict where the player will be in N ticks.
  1384→          const predictedX = playerHead.x + Math.cos(p.angle) * BOT_PREDICT_SPEED * BOT_PREDICT_AHEAD_TICKS;
  1385→          const predictedY = playerHead.y + Math.sin(p.angle) * BOT_PREDICT_SPEED * BOT_PREDICT_AHEAD_TICKS;
  1386→          const distToPredicted = dist(head.x, head.y, predictedX, predictedY);
  1387→
  1388→          // If the bot is on a collision course with the predicted position, evade.
  1389→          if (distToPredicted < (bot.size + p.size) * 3) {
  1390→            // Evade perpendicular to the player's heading.
  1391→            const evadeAngle = p.angle + (Math.random() > 0.5 ? Math.PI / 2 : -Math.PI / 2);
  1392→            desired = evadeAngle;
  1393→            bot.desiredAngle = desired;
  1394→            const wantsBoost = bot.personality === 'hunter' && bot.score > 10;
  1395→            this.tickSnakeMovement(bot, bot.desiredAngle, wantsBoost);
  1396→            return;
  1397→          }
  1398→        }
  1399→      }
  1400→
  1401→      // Also check nearby bot heads for predictive evasion
  1402→      for (const other of this.bots.values()) {
  1403→        if (other.id === bot.id || other.isDead || other.points.length === 0) continue;
  1404→        const otherHead = other.points[0];
  1405→        const otherDist = dist(head.x, head.y, otherHead.x, otherHead.y);
  1406→        if (otherDist < 200) {
  1407→          const predictedX = otherHead.x + Math.cos(other.angle) * BOT_PREDICT_SPEED * BOT_PREDICT_AHEAD_TICKS;
  1408→          const predictedY = otherHead.y + Math.sin(other.angle) * BOT_PREDICT_SPEED * BOT_PREDICT_AHEAD_TICKS;
  1409→          const distToPredicted = dist(head.x, head.y, predictedX, predictedY);
  1410→          if (distToPredicted < (bot.size + other.size) * 3) {
  1411→            const evadeAngle = other.angle + (Math.random() > 0.5 ? Math.PI / 2 : -Math.PI / 2);
  1412→            desired = evadeAngle;
  1413→            bot.desiredAngle = desired;
  1414→            this.tickSnakeMovement(bot, bot.desiredAngle, false);
  1415→            return;
  1416→          }
  1417→        }
  1418→      }
  1419→
  1420→      if (shouldFlee) {
  1421→        desired = Math.atan2(head.y - threatY, head.x - threatX);
  1422→      } else if (bestFood) {
  1423→        desired = Math.atan2(bestFood.y - head.y, bestFood.x - head.x);
  1424→      } else {
  1425→        desired = bot.angle + (Math.random() - 0.5) * 0.4;
  1426→      }
  1427→
  1428→      // NO edge avoidance — infinite map, no boundaries.
  1429→
  1430→      bot.desiredAngle = desired;
  1431→    }
  1432→
  1433→    // Hunter personality occasionally boosts.
  1434→    const wantsBoost = bot.personality === 'hunter' && bot.score > 5 && Math.random() < 0.05;
  1435→    this.tickSnakeMovement(bot, bot.desiredAngle, wantsBoost);
  1436→  }

  1469→  private eatFoodForSnake(snake: SnakeBase): void {
  1470→    if (snake.points.length === 0) return;
  1471→    const head = snake.points[0];
  1472→    const nearby = this.grid.queryRadius(head.x, head.y, snake.collisionRadius + 20);
  1473→    for (const item of nearby.values()) {
  1474→      if (item.kind !== 'food') continue;
  1475→      if ((item.value ?? 0) <= 0) continue;
  1476→      const d = dist(head.x, head.y, item.x, item.y);
  1477→      if (d < snake.collisionRadius + (item.radius || 6) + 6) {
  1478→        // Eat food — add its value to the snake's food-score.
  1479→        const eatenValue = item.value ?? 0;
  1480→        snake.score += eatenValue;
  1481→        item.value = 0;
  1482→        if (item.foodRef) item.foodRef.value = 0;
  1483→        // Spawn small particle burst + sound for player only.
  1484→        if (snake.isPlayer) {
  1485→          const orbSize = eatenValue >= 5 ? 'large' : eatenValue >= 3 ? 'medium' : 'small';
  1486→          playFoodCollect(orbSize);
  1487→          this.spawnEatParticles(item.x, item.y, item.radius || 6, item.foodRef?.color ?? '#fbbf24');
  1488→        }
  1489→      }
  1490→    }
  1491→  }

  1497→  private detectCollisions(now: number): { deadId: string; killerId?: string; cause: 'body' | 'headon' }[] {
  1498→    const deaths: { deadId: string; killerId?: string; cause: 'body' | 'headon' }[] = [];
  1499→    const seenDead = new Set<string>();
  1500→    const all: SnakeBase[] = [];
  1501→    if (this.player && !this.player.isDead) all.push(this.player);
  1502→    for (const bot of this.bots.values()) {
  1503→      if (!bot.isDead) all.push(bot);
  1504→    }
  1505→
  1506→    // --- Head-to-head collision (checked first, takes priority) ---
  1507→    const headOnChecked = new Set<string>();
  1508→    for (let i = 0; i < all.length; i++) {
  1509→      for (let j = i + 1; j < all.length; j++) {
  1510→        const a = all[i];
  1511→        const b = all[j];
  1512→        if (a.isDead || b.isDead) continue;
  1513→        if (now < a.spawnProtectedUntil || now < b.spawnProtectedUntil) continue;
  1514→        const pairKey = a.id < b.id ? `${a.id}:${b.id}` : `${b.id}:${a.id}`;
  1515→        if (headOnChecked.has(pairKey)) continue;
  1516→        headOnChecked.add(pairKey);
  1517→
  1518→        const ha = a.points[0];
  1519→        const hb = b.points[0];
  1520→        const d = dist(ha.x, ha.y, hb.x, hb.y);
  1521→        const hitDist = (a.size + b.size) * this.cfg.headOnHitFactor;
  1522→        if (d >= hitDist) continue;
  1523→
  1524→        // Head-on collision! Determine winner based on rules.
  1525→        const aBoosting = a.isBoosting;
  1526→        const bBoosting = b.isBoosting;
  1527→        const aTotalScore = this.cfg.initialSpawnScore + a.score;
  1528→        const bTotalScore = this.cfg.initialSpawnScore + b.score;
  1529→
  1530→        let aDies = false;
  1531→        let bDies = false;
  1532→
  1533→        if (aTotalScore === bTotalScore) {
  1534→          // Tie — both die
  1535→          aDies = true;
  1536→          bDies = true;
  1537→        } else if (aBoosting && !bBoosting) {
  1538→          // A is boosting, B is not → A wins (boost advantage)
  1539→          bDies = true;
  1540→        } else if (bBoosting && !aBoosting) {
  1541→          // B is boosting, A is not → B wins (boost advantage)
  1542→          aDies = true;
  1543→        } else {
  1544→          // Both boosting or neither boosting → higher score wins
  1545→          if (aTotalScore > bTotalScore) {
  1546→            bDies = true;
  1547→          } else {
  1548→            aDies = true;
  1549→          }
  1550→        }
  1551→
  1552→        if (aDies && !seenDead.has(a.id)) {
  1553→          deaths.push({ deadId: a.id, killerId: b.id, cause: 'headon' });
  1554→          seenDead.add(a.id);
  1555→        }
  1556→        if (bDies && !seenDead.has(b.id)) {
  1557→          deaths.push({ deadId: b.id, killerId: a.id, cause: 'headon' });
  1558→          seenDead.add(b.id);
  1559→        }
  1560→      }
  1561→    }
  1562→
  1563→    // --- Body collision: head vs foreign non-head segment ---
  1564→    for (const snake of all) {
  1565→      if (snake.isDead) continue;
  1566→      if (snake.points.length === 0) continue;
  1567→      if (now < snake.spawnProtectedUntil) continue;
  1568→      if (seenDead.has(snake.id)) continue;
  1569→      const head = snake.points[0];
  1570→
  1571→      // NO wall collision — infinite map.
  1572→
  1573→      const queryR = snake.size + 30;
  1574→      const nearby = this.grid.queryRadius(head.x, head.y, queryR);
  1575→      for (const item of nearby.values()) {
  1576→        if (item.kind !== 'segment') continue;
  1577→        // Own-collision immunity — skip own segments.
  1578→        if (item.snakeId === snake.id) continue;
  1579→        // Skip head segments (handled by head-on collision above).
  1580→        if (item.segIdx === 0) continue;
  1581→        // Neck protection: angle-based — check if approach angle is shallow
  1582→        if (item.segIdx != null && item.segIdx <= 5) continue;
  1583→        const d = dist(head.x, head.y, item.x, item.y);
  1584→        if (d < (snake.size + item.radius) * this.cfg.hitFactor) {
  1585→          deaths.push({ deadId: snake.id, killerId: item.snakeId, cause: 'body' });
  1586→          seenDead.add(snake.id);
  1587→          break;
  1588→        }
  1589→      }
  1590→    }
  1591→    return deaths;
  1592→  }
