    34→// ---------------------------------------------------------------------------
    35→// Password strength calculator
    36→// ---------------------------------------------------------------------------
    37→function getPasswordStrength(pw: string): { label: string; color: string; width: string; score: number } {
    38→  let score = 0;
    39→  if (pw.length >= 6) score++;
    40→  if (pw.length >= 10) score++;
    41→  if (/[A-Z]/.test(pw)) score++;
    42→  if (/[0-9]/.test(pw)) score++;
    43→  if (/[^A-Za-z0-9]/.test(pw)) score++;
    44→
    45→  if (score <= 1) return { label: 'Weak', color: 'bg-red-500', width: 'w-1/4', score };
    46→  if (score <= 2) return { label: 'Fair', color: 'bg-orange-500', width: 'w-2/4', score };
    47→  if (score <= 3) return { label: 'Good', color: 'bg-yellow-500', width: 'w-3/4', score };
    48→  return { label: 'Strong', color: 'bg-emerald-500', width: 'w-full', score };
    49→}

   480→          <button

   485→          >
   486→            {showPassword ? <EyeOff className="w-3.5 h-3.5" /> : <Eye className="w-3.5 h-3.5" />}
   487→          </button>
   488→        </div>
   489→        {/* Password strength indicator */}
   490→        {password.length > 0 && (
   491→          <div className="space-y-1">
   492→            <div className="h-1.5 w-full bg-muted rounded-full overflow-hidden">

   510→            autoComplete="new-password"
   511→            value={confirmPassword}
   512→            onChange={(e) => {
   513→              setConfirmPassword(e.target.value);
   514→              // Clear match error if they now match
   515→              const errEl = document.querySelector('[data-register-error]');
   516→              if (errEl && e.target.value === password) errEl.textContent = '';
   517→            }}
   518→            placeholder="••••••••"
   519→            className="pl-8 pr-9 text-sm"
   520→          />
   521→          <button
   522→            type="button"
   523→            tabIndex={-1}
   524→            className="absolute right-2.5 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground cursor-pointer"
   525→            onClick={() => setShowConfirm(!showConfirm)}
   526→          >
   527→            {showConfirm ? <EyeOff className="w-3.5 h-3.5" /> : <Eye className="w-3.5 h-3.5" />}
   528→          </button>
   529→        </div>
   530→      </div>
   531→      <div className="space-y-1.5">
   532→        <Label htmlFor="r-pin" className="text-xs">Security PIN (4 digits, optional)</Label>
   533→        <Input
   534→          id="r-pin"
   535→          type="text"
   536→          inputMode="numeric"
   537→          maxLength={4}
   538→          pattern="[0-9]{0,4}"
   539→          autoComplete="off"
   540→          value={pin}
   541→          onChange={(e) => setPin(e.target.value.replace(/\D/g, ''))}
   542→          placeholder="e.g. 1234"
   543→          className="text-sm"
   544→        />
   545→        <p className="text-[10px] text-muted-foreground">
   546→          Required for password recovery. Keep it safe!
   547→        </p>
   548→      </div>
   549→
   550→      {error && (
   551→        <p className="text-xs text-destructive flex items-center gap-1" data-register-error>
   552→          <Shield className="w-3 h-3 shrink-0" /> {error}
   553→        </p>
   554→      )}
   555→      {!error && (
   556→        <p className="text-xs text-destructive" data-register-error />
   557→      )}
   558→
   559→      <Button type="submit" className="w-full" disabled={busy}>
   560→        {busy && <Loader2 className="w-4 h-4 mr-2 animate-spin" />}
   561→        Create Account
   562→      </Button>
   563→
   564→      {/* Cross-link to login */}
   565→      <div className="flex justify-center">
   566→        <button
   567→          type="button"
   568→          className="text-[11px] text-primary hover:text-primary/80 underline-offset-2 hover:underline cursor-pointer"
   569→          onClick={() => {
   570→            const tabList = document.querySelector('[role="tablist"]');
   571→            if (tabList) {
   572→              const loginTab = tabList.querySelectorAll('[role="tab"]')[0] as HTMLElement;
   573→              loginTab?.click();
   574→            }
   575→          }}
   576→        >
   577→          Already have an account? <span className="font-semibold">Login</span>
   578→        </button>
   579→      </div>
   580→    </form>
   581→  );
   582→}

   584→// ---------------------------------------------------------------------------
   585→// ForgotPasswordForm (used inside the Dialog)
   586→// ---------------------------------------------------------------------------
   587→function ForgotPasswordForm({
   588→  busy,
   589→  error,
   590→  onSuccess,
   591→}: {
   592→  busy: boolean;
   593→  error: string | null;
   594→  onSuccess: () => void;
   595→}) {
   596→  const [email, setEmail] = useState('');
   597→  const [securityPin, setSecurityPin] = useState('');
   598→  const [newPassword, setNewPassword] = useState('');
   599→  const [confirmNew, setConfirmNew] = useState('');
   600→  const [localError, setLocalError] = useState<string | null>(null);
   601→  const [localBusy, setLocalBusy] = useState(false);
   602→  const [success, setSuccess] = useState(false);
   603→  const [showNewPassword, setShowNewPassword] = useState(false);
   604→
   605→  async function handleSubmit(e: React.FormEvent) {
   606→    e.preventDefault();
   607→    setLocalError(null);
   608→
   609→    if (newPassword !== confirmNew) {
   610→      setLocalError('Passwords do not match.');
   611→      return;
   612→    }
   613→
   614→    setLocalBusy(true);
   615→    try {
   616→      const res = await fetch('/api/auth/forgot-password', {
   617→        method: 'POST',
   618→        headers: { 'Content-Type': 'application/json' },
   619→        body: JSON.stringify({ email, securityPin, newPassword }),
   620→      });
   621→      const data = await res.json().catch(() => ({}));
   622→      if (!res.ok) {
   623→        setLocalError(data?.error || 'Failed to reset password.');
   624→        return;
   625→      }
   626→      setSuccess(true);
   627→      onSuccess();
   628→    } catch {
   629→      setLocalError('Network error. Please try again.');
   630→    } finally {
   631→      setLocalBusy(false);
   632→    }
   633→  }
   634→
   635→  if (success) {
   636→    return (
   637→      <div className="text-center py-4 space-y-2">
   638→        <div className="w-12 h-12 rounded-full bg-emerald-500/10 border border-emerald-500/20 flex items-center justify-center mx-auto">
   639→          <Shield className="w-6 h-6 text-emerald-400" />
   640→        </div>
   641→        <p className="text-sm font-semibold text-foreground">Password Reset!</p>
   642→        <p className="text-xs text-muted-foreground">
   643→          Your password has been changed. You can now log in with your new password.
   644→        </p>
   645→        <Button
   646→          size="sm"
   647→          className="mt-2"
   648→          onClick={onSuccess}
   649→        >
   650→          Back to Login
   651→        </Button>
   652→      </div>
   653→    );
   654→  }
   655→
   656→  return (
   657→    <form onSubmit={handleSubmit} className="space-y-3">
   658→      <div className="space-y-1.5">
   659→        <Label htmlFor="fp-email" className="text-xs">Email</Label>
   660→        <Input
   661→          id="fp-email"
   662→          type="email"
   663→          required
   664→          value={email}
   665→          onChange={(e) => setEmail(e.target.value)}
   666→          placeholder="you@arena.gg"
   667→          className="text-sm"
   668→        />
   669→      </div>
   670→      <div className="space-y-1.5">
   671→        <Label htmlFor="fp-pin" className="text-xs">4-Digit Security PIN</Label>
   672→        <Input
   673→          id="fp-pin"
   674→          type="text"
   675→          inputMode="numeric"
   676→          required
   677→          maxLength={4}
   678→          pattern="[0-9]{4}"
   679→          autoComplete="off"
   680→          value={securityPin}
   681→          onChange={(e) => setSecurityPin(e.target.value.replace(/\D/g, ''))}
   682→          placeholder="1234"
   683→          className="text-sm"
   684→        />
   685→        <p className="text-[10px] text-muted-foreground">
   686→          This is the PIN you set during registration.
   687→        </p>
   688→      </div>
   689→      <div className="space-y-1.5">
   690→        <Label htmlFor="fp-new" className="text-xs">New Password (min 6 chars)</Label>
   691→        <div className="relative">
   692→          <Input
   693→            id="fp-new"
   694→            type={showNewPassword ? 'text' : 'password'}
   695→            required
   696→            minLength={6}
   697→            value={newPassword}
   698→            onChange={(e) => setNewPassword(e.target.value)}
   699→            placeholder="••••••••"
   700→            className="pr-9 text-sm"
   701→          />
   702→          <button
   703→            type="button"
   704→            tabIndex={-1}
   705→            className="absolute right-2.5 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground cursor-pointer"
   706→            onClick={() => setShowNewPassword(!showNewPassword)}
   707→          >
   708→            {showNewPassword ? <EyeOff className="w-3.5 h-3.5" /> : <Eye className="w-3.5 h-3.5" />}
   709→          </button>
   710→        </div>
   711→      </div>
   712→      <div className="space-y-1.5">
   713→        <Label htmlFor="fp-confirm" className="text-xs">Confirm New Password</Label>
   714→        <Input
   715→          id="fp-confirm"
   716→          type="password"
   717→          required
   718→          minLength={6}
   719→          value={confirmNew}
   720→          onChange={(e) => {
   721→            setConfirmNew(e.target.value);
   722→            if (e.target.value === newPassword) setLocalError(null);
   723→          }}
   724→          placeholder="••••••••"
   725→          className="text-sm"
   726→        />
   727→      </div>
   728→
   729→      {(localError || error) && (
   730→        <p className="text-xs text-destructive">{localError || error}</p>
   731→      )}
   732→
   733→      <Button type="submit" className="w-full" disabled={localBusy || busy}>
   734→        {(localBusy || busy) && <Loader2 className="w-4 h-4 mr-2 animate-spin" />}
   735→        Reset Password
   736→      </Button>
   737→    </form>
   738→  );
   739→}
