     8→/** Pick N random items from an array (no duplicates) */
     9→function pickRandom<T>(arr: T[], n: number): T[] {
    10→  const shuffled = [...arr].sort(() => Math.random() - 0.5);
    11→  return shuffled.slice(0, n);
    12→}
    13→
    14→/** Pick N random items ensuring no duplicate categories */
    15→function pickDiverse<T extends { category: string }>(arr: T[], n: number): T[] {
    16→  // Group by category
    17→  const byCategory = new Map<string, T[]>();
    18→  for (const item of arr) {
    19→    const list = byCategory.get(item.category) || [];
    20→    list.push(item);
    21→    byCategory.set(item.category, list);
    22→  }
    23→
    24→  // Shuffle categories
    25→  const categories = [...byCategory.keys()].sort(() => Math.random() - 0.5);
    26→  const picked: T[] = [];
    27→  const usedCategories = new Set<string>();
    28→
    29→  // First pass: pick one from each unique category
    30→  for (const cat of categories) {
    31→    if (picked.length >= n) break;
    32→    const items = byCategory.get(cat)!;
    33→    const item = items[Math.floor(Math.random() * items.length)];
    34→    picked.push(item);
    35→    usedCategories.add(cat);
    36→  }
    37→
    38→  // Second pass: fill remaining slots (still avoiding duplicates in the pick)
    39→  if (picked.length < n) {
    40→    const remaining = arr.filter(
    41→      (item) => !picked.includes(item),
    42→    );
    43→    const extra = pickRandom(remaining, n - picked.length);
    44→    picked.push(...extra);
    45→  }
    46→
    47→  return picked.slice(0, n);
    48→}

    50→/** Exclude templates that share a title with the exclude list */
    51→function excludeByTitle<T extends { title: string }>(pool: T[], excludeTitles: Set<string>): T[] {
    52→  return pool.filter((t) => !excludeTitles.has(t.title));

   271→
   272→/** Count consecutive days where ALL daily challenges were claimed */
   273→async function calculateStreak(playerId: string): Promise<{ streak: number; multiplier: number }> {
   274→  const today = utcToday();
   275→
   276→  // Compute the date 30 days ago as YYYY-MM-DD
   277→  const d = new Date(today + 'T00:00:00Z');
   278→  d.setUTCDate(d.getUTCDate() - 29);
   279→  const thirtyDaysAgo = `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, '0')}-${String(d.getUTCDate()).padStart(2, '0')}`;
   280→
   281→  // Fetch ALL daily challenges for the last 30 days in a single query
   282→  const recentChallenges = await db.challenge.findMany({
   283→    where: {
   284→      playerId,
   285→      type: 'daily',
   286→      periodStart: { gte: thirtyDaysAgo },
   287→    },
   288→    select: { periodStart: true, claimed: true },
   289→  });
   290→
   291→  // Group by date
   292→  const byDate = new Map<string, boolean[]>();
   293→  for (const c of recentChallenges) {
   294→    const list = byDate.get(c.periodStart) ?? [];
   295→    list.push(c.claimed);
   296→    byDate.set(c.periodStart, list);
   297→  }
   298→
   299→  // Walk backwards from today checking consecutive claimed days
   300→  let streak = 0;
   301→  for (let i = 0; i < 30; i++) {
   302→    const checkDate = new Date(today + 'T00:00:00Z');
   303→    checkDate.setUTCDate(checkDate.getUTCDate() - i);
   304→    const key = `${checkDate.getUTCFullYear()}-${String(checkDate.getUTCMonth() + 1).padStart(2, '0')}-${String(checkDate.getUTCDate()).padStart(2, '0')}`;
   305→
   306→    const dayClaims = byDate.get(key);
   307→    if (!dayClaims || dayClaims.length === 0) {
   308→      // No challenges for this day
   309→      if (i === 0) continue; // today may not have challenges yet
   310→      break; // gap found — streak ends
   311→    }
   312→
   313→    const allClaimed = dayClaims.every(Boolean);
   314→    if (allClaimed) {
   315→      streak++;
   316→    } else if (i === 0) {
   317→      // Today's challenges exist but not all claimed — don't break, just don't count today
   318→      continue;
   319→    } else {
   320→      break; // unclaimed challenges on a past day — streak broken
   321→    }
   322→  }
   323→
   324→  // Calculate multiplier
   325→  let multiplier = 1.0;
   326→  if (streak >= 14) multiplier = 3.0;
   327→  else if (streak >= 7) multiplier = 2.0;
   328→  else if (streak >= 3) multiplier = 1.5;
   329→
   330→  return { streak, multiplier };
   331→}

   333→// ---------------------------------------------------------------------------
   334→// GET handler — fetch (and auto-generate) challenges
   335→// ---------------------------------------------------------------------------
   336→
   337→export async function GET() {
   338→  const session = await getSession();
   339→  if (!session) {
   340→    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
   341→  }
   342→
   343→  const playerId = session.playerId;
   344→  const today = utcToday();
   345→  const monday = utcMonday();
   346→
   347→  // Get player level for tier-based pool selection
   348→  const player = await db.player.findUnique({ where: { id: playerId } });
   349→  if (!player) {
   350→    return NextResponse.json({ error: 'Player not found.' }, { status: 404 });
   351→  }
   352→
   353→  const tier = getLevelTier(player.level);
   354→  const rewardMult = levelRewardMultiplier(player.level);
   355→
   356→  // ── Anti-repeat: fetch yesterday's dailies and last week's weeklies ──
   357→  const yesterday = utcYesterday();
   358→  const lastMonday = utcLastMonday();
   359→
   360→  const [yesterdayDailies, lastWeekWeeklies] = await Promise.all([
   361→    db.challenge.findMany({
   362→      where: { playerId, type: 'daily', periodStart: yesterday },
   363→      select: { title: true },
   364→    }),
   365→    db.challenge.findMany({
   366→      where: { playerId, type: 'weekly', periodStart: lastMonday },
   367→      select: { title: true },
   368→    }),
   369→  ]);
   370→
   371→  const excludeDailyTitles = new Set(yesterdayDailies.map((c) => c.title));
   372→  const excludeWeeklyTitles = new Set(lastWeekWeeklies.map((c) => c.title));
   373→
   374→  // Check existing daily challenges for today
   375→  const existingDaily = await db.challenge.findMany({
   376→    where: { playerId, type: 'daily', periodStart: today },
   377→  });
   378→
   379→  // Check existing weekly challenges for this week
   380→  const existingWeekly = await db.challenge.findMany({
   381→    where: { playerId, type: 'weekly', periodStart: monday },
   382→  });
   383→
   384→  // ── Generate daily challenges if needed ──
   385→  if (existingDaily.length === 0) {
   386→    let pool = DAILY_POOLS[tier];
   387→    // Exclude yesterday's titles (anti-repeat)
   388→    pool = excludeByTitle(pool, excludeDailyTitles);
   389→    // Fallback: if pool is too small after exclusions, use full pool
   390→    if (pool.length < 3) pool = DAILY_POOLS[tier];
   391→    // Pick 3 diverse challenges (different categories guaranteed)
   392→    const templates = pickDiverse(pool, 3);
   393→
   394→    await db.challenge.createMany({
   395→      data: templates.map((t) => ({
   396→        playerId,
   397→        type: 'daily',
   398→        category: t.category,
   399→        title: t.title,
   400→        description: t.description,
   401→        target: t.target,
   402→        reward: Math.floor(t.reward * rewardMult),
   403→        periodStart: today,
   404→      })),
   405→    });
   406→  }
   407→
   408→  // ── Generate weekly challenges if needed ──
   409→  if (existingWeekly.length === 0) {
   410→    let pool = WEEKLY_POOLS[tier];
   411→    // Exclude last week's titles (anti-repeat)
   412→    pool = excludeByTitle(pool, excludeWeeklyTitles);
   413→    // Fallback: if pool is too small after exclusions, use full pool
   414→    if (pool.length < 2) pool = WEEKLY_POOLS[tier];
   415→    // Pick 2 diverse challenges (different categories guaranteed)
   416→    const templates = pickDiverse(pool, 2);
   417→
   418→    await db.challenge.createMany({
   419→      data: templates.map((t) => ({
   420→        playerId,
   421→        type: 'weekly',
   422→        category: t.category,
   423→        title: t.title,
   424→        description: t.description,
   425→        target: t.target,
   426→        reward: Math.floor(t.reward * rewardMult),
   427→        periodStart: monday,
   428→      })),
   429→    });
   430→  }
   431→
   432→  // Fetch all active challenges (today's daily + this week's weekly)
   433→  const challenges = await db.challenge.findMany({
   434→    where: {
   435→      playerId,
   436→      OR: [
   437→        { type: 'daily', periodStart: today },
   438→        { type: 'weekly', periodStart: monday },
   439→      ],
   440→    },
   441→    orderBy: [{ type: 'asc' }, { createdAt: 'asc' }],
   442→  });
   443→
   444→  // Calculate streak info
   445→  const streakInfo = await calculateStreak(playerId);
   446→
   447→  return NextResponse.json({
   448→    challenges,
   449→    streak: streakInfo.streak,
   450→    streakMultiplier: streakInfo.multiplier,
   451→    tier,
   452→  });
   453→}

   455→// ---------------------------------------------------------------------------
   456→// POST handler — claim a completed challenge (with streak bonus)
   457→// ---------------------------------------------------------------------------
   458→
   459→export async function POST(req: NextRequest) {
   460→  const session = await getSession();
   461→  if (!session) {
   462→    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
   463→  }
   464→
   465→  const body = await req.json().catch(() => ({}));
   466→  const challengeId = String(body.challengeId || '').trim();
   467→
   468→  if (!challengeId) {
   469→    return NextResponse.json({ error: 'Missing challengeId.' }, { status: 400 });
   470→  }
   471→
   472→  const playerId = session.playerId;
   473→
   474→  // Find the challenge and verify ownership
   475→  const challenge = await db.challenge.findUnique({
   476→    where: { id: challengeId },
   477→  });
   478→
   479→  if (!challenge || challenge.playerId !== playerId) {
   480→    return NextResponse.json({ error: 'Challenge not found.' }, { status: 404 });
   481→  }
   482→
   483→  if (!challenge.completed) {
   484→    return NextResponse.json({ error: 'Challenge not yet completed.' }, { status: 400 });
   485→  }
   486→
   487→  if (challenge.claimed) {
   488→    return NextResponse.json({ error: 'Already claimed.' }, { status: 400 });
   489→  }
   490→
   491→  // Calculate streak bonus
   492→  const { multiplier } = await calculateStreak(playerId);
   493→  const baseReward = challenge.reward;
   494→  const bonusReward = Math.floor(baseReward * (multiplier - 1)); // bonus = base × (mult - 1)
   495→  const totalReward = baseReward + bonusReward;
   496→
   497→  // Credit reward chips and mark as claimed (atomic transaction)
   498→  await db.$transaction([
   499→    db.player.update({
   500→      where: { id: playerId },
   501→      data: {
   502→        bankedChips: { increment: totalReward },
   503→        totalEarned: { increment: totalReward },
   504→      },
   505→    }),
   506→    db.challenge.update({
   507→      where: { id: challengeId },
   508→      data: { claimed: true },
   509→    }),
   510→  ]);
   511→
   512→  return NextResponse.json({
   513→    success: true,
   514→    reward: totalReward,
   515→    baseReward,
   516→    bonusReward,
   517→    streakMultiplier: multiplier,
   518→  });
   519→}
