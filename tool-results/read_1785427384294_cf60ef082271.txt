    46→function buildMatchHistory(p: InspectedPlayer): MatchLog[] {
    47→  const bigChip = p.bankedChips >= 10_000_000 ? 10_000_000 : 2_500_000;
    48→  return [
    49→    { arena: 'Tier-05 Crore High Roller', outcome: 'Extracted', chips: bigChip, time: '10 mins ago', kills: 14 },
    50→    { arena: 'Tier-04 Platinum Arena', outcome: 'Extracted', chips: 1_500_000, time: '2 hours ago', kills: 8 },
    51→    { arena: 'Tier-03 Viper Boundary', outcome: 'Extracted', chips: 500_000, time: '1 day ago', kills: 5 },
    52→    { arena: 'Tier-05 Crore High Roller', outcome: 'Eliminated', chips: -200_000, time: '2 days ago', kills: 3 },
    53→  ];
    54→}

   438→                    <div className="font-bold text-amber-400 text-sm mt-0.5">#{globalRank}</div>
   439→                  </div>
   440→                  <div className="p-2 bg-slate-900 rounded-lg border border-slate-800">
   441→                    <div className="text-[9px] text-slate-400">{flag} Country Rank</div>
   442→                    <div className="font-bold text-emerald-400 text-sm mt-0.5">#{countryRank}</div>
   443→                  </div>
   444→                  <div className="p-2 bg-slate-900 rounded-lg border border-slate-800">
   445→                    <div className="text-[9px] text-slate-400">Regional Arena Rank</div>
   446→                    <div className="font-bold text-indigo-400 text-sm mt-0.5">#{regionalRank}</div>
   447→                  </div>
   448→                </div>
   449→              </div>
   450→              <div className="grid grid-cols-2 gap-3">
   451→                <StatCard label="Total Banked Chips" value={`${p.bankedChips.toLocaleString('en-IN')} c`} accent="text-emerald-400" icon={<Trophy className="w-3.5 h-3.5" />} />
   452→                <StatCard label="Highest Extraction" value={highestExtraction} accent="text-amber-400" icon={<Award className="w-3.5 h-3.5" />} />
   453→                <StatCard label="Extraction Success Rate" value={successRate} accent="text-indigo-400" icon={<Zap className="w-3.5 h-3.5" />} />
   454→                <StatCard label="Snake Eliminations" value={totalKills} accent="text-rose-400" icon={<Swords className="w-3.5 h-3.5" />} />
   455→              </div>
   456→              {/* Additional real stats when available */}
   457→              {(p.lifetimeExtracts != null || p.bestStreak != null) && (
   458→                <div className="grid grid-cols-2 gap-3">
   459→                  {p.lifetimeExtracts != null && (
   460→                    <StatCard
   461→                      label="Total Extractions"
   462→                      value={p.lifetimeExtracts.toLocaleString()}
   463→                      accent="text-cyan-400"
   464→                      icon={<Sparkles className="w-3.5 h-3.5" />}
   465→                    />
   466→                  )}
   467→                  {p.bestStreak != null && (
   468→                    <StatCard
   469→                      label="Best Streak"
   470→                      value={`${p.bestStreak} Wins`}
   471→                      accent="text-yellow-400"
   472→                      icon={<Trophy className="w-3.5 h-3.5" />}
   473→                    />
   474→                  )}
   475→                </div>
   476→              )}
   477→            </div>
   478→          )}
   479→
   480→          {/* EXTRACTION LOGS */}
   481→          {tab === 'logs' && (
   482→            <div className="space-y-2">
   483→              {history.map((log, i) => (
   484→                <div key={i} className="p-3 bg-slate-950 rounded-xl border border-slate-800 flex items-center justify-between text-xs gap-2">
   485→                  <div className="min-w-0">
   486→                    <div className="font-bold text-white flex items-center gap-1.5 flex-wrap">
   487→                      <span className="truncate">{log.arena}</span>
   488→                      <span className={`text-[9px] font-mono font-bold px-1.5 py-0.2 rounded ${log.outcome === 'Extracted' ? 'bg-emerald-500/20 text-emerald-300' : 'bg-rose-500/20 text-rose-300'}`}>
   489→                        {log.outcome}
   490→                      </span>
   491→                    </div>
   492→                    <div className="text-[10px] text-slate-400 mt-0.5">
   493→                      🕒 {log.time} · {log.kills} kills
   494→                    </div>
   495→                  </div>
   496→                  <div className={`font-mono font-bold tabular-nums shrink-0 ${log.chips > 0 ? 'text-emerald-400' : 'text-rose-400'}`}>
   497→                    {log.chips > 0 ? `+${log.chips.toLocaleString('en-IN')}c` : `${log.chips.toLocaleString('en-IN')}c`}
   498→                  </div>
   499→                </div>
   500→              ))}
   501→            </div>
   502→          )}
   503→
   504→          {/* LOADOUT */}
   505→          {tab === 'loadout' && (
   506→            <div className="bg-slate-950 p-4 rounded-xl border border-slate-800 space-y-2 text-xs">
   507→              {loadoutEntries.map((l) => (
   508→                <div key={l.label} className="flex justify-between py-1.5 border-b border-slate-900 last:border-0">
   509→                  <span className="text-slate-400">{l.label}</span>
   510→                  <span className={`font-bold ${l.value === 'Not visible' ? 'text-slate-500 italic' : 'text-white'}`}>{l.value}</span>
   511→                </div>
   512→              ))}
   513→            </div>
   514→          )}
   515→        </div>
   516→
   517→        {/* Action buttons */}
   518→        <div className="grid grid-cols-2 gap-2 p-4 pt-2 border-t border-slate-800 shrink-0">
   519→          <button
   520→            type="button"
   521→            onClick={handleAddFriend}
   522→            disabled={friendRequested}
   523→            className={`py-2.5 px-3 border rounded-xl text-xs font-bold flex items-center justify-center gap-1.5 transition-all ${friendRequested ? 'bg-emerald-500/20 text-emerald-400 border-emerald-500/40 cursor-default' : 'bg-emerald-600 hover:bg-emerald-500 text-white border-emerald-500/30 shadow'}`}
   524→          >
   525→            {friendRequested ? <Check className="w-4 h-4" /> : <UserPlus className="w-4 h-4" />}
   526→            {friendRequested ? 'Request Sent' : 'Add Friend'}
   527→          </button>
   528→          <button
   529→            type="button"
   530→            onClick={handleChallenge}
   531→            className="py-2.5 px-3 bg-indigo-600 hover:bg-indigo-500 text-white border border-indigo-500/30 rounded-xl text-xs font-bold flex items-center justify-center gap-1.5 transition shadow"
   532→          >
   533→            <Swords className="w-4 h-4" /> Challenge
   534→          </button>
   535→          <button
   536→            type="button"
   537→            onClick={handleBlock}
   538→            disabled={blocked}
   539→            className={`col-span-2 py-2 border rounded-xl text-xs font-bold flex items-center justify-center gap-1.5 transition-all ${blocked ? 'bg-rose-950/60 text-slate-400 border-slate-800 cursor-default' : 'bg-rose-950/20 hover:bg-rose-950/40 text-rose-400 border-rose-500/20'}`}
   540→          >
   541→            <Ban className="w-3.5 h-3.5" /> {blocked ? 'Player Blocked' : 'Block Player'}
   542→          </button>
   543→        </div>
   544→      </div>
   545→    </div>
   546→  );
   547→}
   548→
   549→function StatCard({ label, value, accent, icon }: { label: string; value: string; accent: string; icon: React.ReactNode }) {
   550→  return (
   551→    <div className="bg-slate-950 p-3 rounded-xl border border-slate-800 space-y-1">

   553→        {icon} {label}
   554→      </span>
   555→      <span className={`font-mono font-bold text-sm block ${accent}`}>{value}</span>
   556→    </div>
   557→  );
   558→}
