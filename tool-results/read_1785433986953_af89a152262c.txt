    82→    82→/** Returns the world-space rectangle visible in the viewport, plus margin. */
    83→    83→export function computeVisibleRect(rc: FrameRenderCtx, marginPx = 100): VisibleRect {
    84→    84→  const halfW = rc.w / 2 / rc.zoom;
    85→    85→  const halfH = rc.h / 2 / rc.zoom;
    86→    86→  const m = marginPx / rc.zoom;
    87→    87→  return {
    88→    88→    left: rc.camX - halfW - m,
    89→    89→    right: rc.camX + halfW + m,
    90→    90→    top: rc.camY - halfH - m,
    91→    91→    bottom: rc.camY + halfH + m,
    92→    92→  };
    93→    93→}

    95→    95→/** Quick AABB-vs-rect test. */
    96→    96→function rectContainsPoint(rect: VisibleRect, x: number, y: number): boolean {
    97→    97→  return x >= rect.left && x <= rect.right && y >= rect.top && y <= rect.bottom;
    98→    98→}

   100→   100→/** Returns true if any point in the list is inside the visible rect. */
   101→   101→function snakeIsVisible(pts: { x: number; y: number }[], rect: VisibleRect): boolean {
   102→   102→  if (pts.length === 0) return false;
   103→   103→  if (rectContainsPoint(rect, pts[0].x, pts[0].y)) return true;
   104→   104→  const last = pts[pts.length - 1];
   105→   105→  if (rectContainsPoint(rect, last.x, last.y)) return true;
   106→   106→  const stride = pts.length > 60 ? 4 : 2;
   107→   107→  for (let i = 0; i < pts.length; i += stride) {
   108→   108→    if (rectContainsPoint(rect, pts[i].x, pts[i].y)) return true;
   109→   109→  }
   110→   110→  return false;
   111→   111→}

   113→   113→/**
   114→   114→ * Cache a metallic radial gradient. Bucketed by 2-pixel radius so we don't
   115→   115→ * cache a gradient for every floating-point size value.
   116→   116→ */
   117→   117→function getMetallicGradient(
   118→   118→  rc: FrameRenderCtx,
   119→   119→  radius: number,
   120→   120→  color: string,
   121→   121→  secondary: string | undefined,
   122→   122→): CanvasGradient | string {
   123→   123→  const bucket = Math.max(4, Math.round(radius));
   124→   124→  const key = `${color}:${secondary ?? ''}:${bucket}`;
   125→   125→  const cached = rc.metallicCache.get(key);
   126→   126→  if (cached) return cached;
   127→   127→  const { ctx } = rc;
   128→   128→  const g = ctx.createRadialGradient(-bucket * 0.35, -bucket * 0.35, bucket * 0.1, 0, 0, bucket);
   129→   129→  g.addColorStop(0, '#f8fafc');
   130→   130→  g.addColorStop(0.35, secondary || color);
   131→   131→  g.addColorStop(1, color);
   132→   132→  rc.metallicCache.set(key, g);
   133→   133→  return g;
   134→   134→}

   136→   136→/**
   137→   137→ * Breathing arena radius at a given time. Matches the server's
   138→   138→ * `getMapRadius(elapsedMs)` formula exactly:
   139→   139→ *   baseRadius + sin(cycleProgress * 2π) * amplitude
   140→   140→ */
   141→   141→export function getArenaRadius(now: number): number {
   142→   142→  const cycleTime = (now % MAP_BREATH_CYCLE_MS) / MAP_BREATH_CYCLE_MS;
   143→   143→  const sinVal = Math.sin(cycleTime * Math.PI * 2);
   144→   144→  return MAP_BASE_RADIUS + sinVal * MAP_BREATH_AMPLITUDE;
   145→   145→}

   148→   148→// Indian number formatting for chip display
   149→   149→// ---------------------------------------------------------------------------
   150→   150→
   151→   151→/**
   152→   152→ * Formats a chip count using Indian numbering system.
   153→   153→ *  - Under 1000: plain number (e.g. "500")
   154→   154→ *  - 1K to 99K: "1.5K", "25K"
   155→   155→ *  - 1L (1,00,000) to 99L: "1L", "50L"
   156→   156→ *  - 1Cr (1,00,00,000)+: "1.2Cr", "15Cr"
   157→   157→ */
   158→   158→export function formatChipDisplay(chips: number): string {
   159→   159→  if (chips < 1000) return String(chips);
   160→   160→  if (chips < 100_000) {
   161→   161→    // K range: 1K to 99.9K
   162→   162→    const k = chips / 1000;
   163→   163→    return k % 1 === 0 ? `${k}K` : `${k.toFixed(1).replace(/\.0$/, '')}K`;
   164→   164→  }
   165→   165→  if (chips < 10_000_000) {
   166→   166→    // L (lakh) range: 1L to 99.9L
   167→   167→    const l = chips / 100_000;
   168→   168→    return l % 1 === 0 ? `${l}L` : `${l.toFixed(1).replace(/\.0$/, '')}L`;
   169→   169→  }
   170→   170→  // Cr (crore) range
   171→   171→  const cr = chips / 10_000_000;
   172→   172→  return cr % 1 === 0 ? `${cr}Cr` : `${cr.toFixed(1).replace(/\.0$/, '')}Cr`;
   173→   173→}

   176→   176→// Grid + world bounds

   179→   179→/**
   180→   180→ * Draws the breathing circular arena: background fill (deep slate `#020617`)
   181→   181→ * clipped to the circle, a subtle grid (`#1e293b`, gridSize 60) inside the
   182→   182→ * circle, and the neon-rose boundary (`#f43f5e`, lineWidth 10, shadowBlur 16).
   183→   183→ *
   184→   184→ * The radius breathes using the same sinusoidal formula as the server.
   185→   185→ */
   186→   186→export function drawGrid(rc: FrameRenderCtx): void {
   187→   187→  const { ctx, worldSize, camX, camY, zoom, w, h } = rc;
   188→   188→  const cx = worldSize / 2;
   189→   189→  const cy = worldSize / 2;
   190→   190→  const radius = getArenaRadius(rc.now);
   191→   191→
   192→   192→  // --- Background fill (entire arena disc) ---
   193→   193→  ctx.save();
   194→   194→  ctx.beginPath();
   195→   195→  ctx.arc(cx, cy, radius, 0, Math.PI * 2);
   196→   196→  ctx.fillStyle = '#020617';
   197→   197→  ctx.fill();
   198→   198→  ctx.clip();
   199→   199→
   200→   200→  // --- Grid lines (subtle slate-800, gridSize 60) ---
   201→   201→  const grid = 60;
   202→   202→  const vis = computeVisibleRect(rc);
   203→   203→  const startX = Math.max(0, Math.floor(vis.left / grid) * grid);
   204→   204→  const endX = Math.min(worldSize, Math.ceil(vis.right / grid) * grid);
   205→   205→  const startY = Math.max(0, Math.floor(vis.top / grid) * grid);
   206→   206→  const endY = Math.min(worldSize, Math.ceil(vis.bottom / grid) * grid);
   207→   207→
   208→   208→  ctx.strokeStyle = '#1e293b';
   209→   209→  ctx.lineWidth = 1 / zoom;
   210→   210→  ctx.beginPath();
   211→   211→  for (let x = startX; x <= endX; x += grid) {
   212→   212→    ctx.moveTo(x, startY);
   213→   213→    ctx.lineTo(x, endY);
   214→   214→  }
   215→   215→  for (let y = startY; y <= endY; y += grid) {
   216→   216→    ctx.moveTo(startX, y);
   217→   217→    ctx.lineTo(endX, y);
   218→   218→  }
   219→   219→  ctx.stroke();
   220→   220→  ctx.restore(); // remove clip
   221→   221→
   222→   222→  // --- Outer boundary (neon rose, breathing) ---
   223→   223→  ctx.save();
   224→   224→  ctx.lineWidth = 10;
   225→   225→  ctx.strokeStyle = '#f43f5e';
   226→   226→  if (!rc.lowQuality) {
   227→   227→    ctx.shadowColor = '#f43f5e';
   228→   228→    ctx.shadowBlur = 16;
   229→   229→  }
   230→   230→  ctx.beginPath();
   231→   231→  ctx.arc(cx, cy, radius, 0, Math.PI * 2);
   232→   232→  ctx.stroke();
   233→   233→  ctx.shadowBlur = 0;
   234→   234→  ctx.restore();
   235→   235→
   236→   236→  // Silence unused-param lint for camX/camY/w/h.
   237→   237→  void camX;
   238→   238→  void camY;
   239→   239→  void w;
   240→   240→  void h;
   241→   241→}

   244→   244→// Dynamic map boundary (for online games with dynamic radius)
   245→   245→// ---------------------------------------------------------------------------
   246→   246→
   247→   247→/**
   248→   248→ * Draws a circular map boundary with neon-rose glow and a subtle breathing
   249→   249→ * pulse. Used for online arenas where `mapRadius` comes from the GameSnapshot
   250→   250→ * rather than the fixed breathing formula.
   251→   251→ */
   252→   252→export function drawMapBoundary(
   253→   253→  ctx: CanvasRenderingContext2D,
   254→   254→  centerX: number,
   255→   255→  centerY: number,
   256→   256→  radius: number,
   257→   257→  tick: number,
   258→   258→): void {
   259→   259→  // Subtle breathing pulse: ±3px oscillation over a 4-second cycle
   260→   260→  const breathe = Math.sin(tick * 0.0015) * 3;
   261→   261→  const r = Math.max(100, radius + breathe);
   262→   262→
   263→   263→  ctx.save();
   264→   264→  // Outer glow
   265→   265→  ctx.lineWidth = 10;
   266→   266→  ctx.strokeStyle = '#f43f5e';
   267→   267→  ctx.shadowColor = '#f43f5e';
   268→   268→  ctx.shadowBlur = 20;
   269→   269→  ctx.beginPath();
   270→   270→  ctx.arc(centerX, centerY, r, 0, Math.PI * 2);
   271→   271→  ctx.stroke();
   272→   272→
   273→   273→  // Inner subtle glow ring
   274→   274→  ctx.shadowBlur = 8;
   275→   275→  ctx.lineWidth = 3;
   276→   276→  ctx.strokeStyle = 'rgba(244, 63, 94, 0.35)';
   277→   277→  ctx.beginPath();
   278→   278→  ctx.arc(centerX, centerY, r - 6, 0, Math.PI * 2);
   279→   279→  ctx.stroke();
   280→   280→
   281→   281→  ctx.shadowBlur = 0;
   282→   282→  ctx.restore();
   283→   283→}

   288→   288→
   289→   289→/** Orb type configuration for the three visual tiers. */
   290→   290→interface OrbConfig {
   291→   291→  radius: number;
   292→   292→  color: string;
   293→   293→  glowColor: string;
   294→   294→  shadowBlur: number;
   295→   295→  label: 'small' | 'medium' | 'large';
   296→   296→}
   297→   297→
   298→   298→const ORB_CONFIGS: Record<number, OrbConfig> = {
   299→   299→  1: { radius: 3, color: '#34d399', glowColor: '#10b981', shadowBlur: 6, label: 'small' },
   300→   300→  3: { radius: 5, color: '#38bdf8', glowColor: '#0ea5e9', shadowBlur: 10, label: 'medium' },
   301→   301→  5: { radius: 8, color: '#f472b6', glowColor: '#ec4899', shadowBlur: 16, label: 'large' },
   302→   302→};
   303→   303→
   304→   304→/** Default config used when orbSize doesn't match known values. */
   305→   305→const DEFAULT_ORB: OrbConfig = ORB_CONFIGS[1];
   306→   306→
   307→   307→/**
   308→   308→ * Draws a single food orb with glow effects. The orbSize determines which
   309→   309→ * visual tier to use. Falls back to small orb for unknown sizes.
   310→   310→ */
   311→   311→export function drawFoodOrb(
   312→   312→  ctx: CanvasRenderingContext2D,
   313→   313→  x: number,
   314→   314→  y: number,
   315→   315→  orbSize: number,
   316→   316→  value: number,
   317→   317→  color: string,
   318→   318→  glowColor: string,
   319→   319→  now: number,
   320→   320→  lowQuality: boolean,
   321→   321→): void {
   322→   322→  // Select config based on value; use provided color/glowColor if different
   323→   323→  const config = ORB_CONFIGS[value] ?? DEFAULT_ORB;
   324→   324→  const useColor = color || config.color;
   325→   325→  const useGlow = glowColor || config.glowColor;
   326→   326→  const r = Math.max(2, orbSize > 0 ? orbSize : config.radius);
   327→   327→
   328→   328→  ctx.save();
   329→   329→
   330→   330→  // Large orbs get a subtle pulse animation
   331→   331→  let animR = r;
   332→   332→  if (value === 5 && !lowQuality) {
   333→   333→    const pulse = Math.sin(now * 0.004) * 1.5;
   334→   334→    animR = r + pulse;
   335→   335→  }
   336→   336→
   337→   337→  // Glow effect (skip in low quality)
   338→   338→  if (!lowQuality) {
   339→   339→    ctx.shadowColor = useGlow;
   340→   340→    ctx.shadowBlur = config.shadowBlur;
   341→   341→  }
   342→   342→
   343→   343→  // Radial gradient for a lit-from-within look
   344→   344→  const grad = ctx.createRadialGradient(x, y, 0, x, y, animR);
   345→   345→  grad.addColorStop(0, '#ffffff');
   346→   346→  grad.addColorStop(0.3, useColor);
   347→   347→  grad.addColorStop(1, useGlow);
   348→   348→
   349→   349→  ctx.beginPath();
   350→   350→  ctx.arc(x, y, animR, 0, Math.PI * 2);
   351→   351→  ctx.fillStyle = grad;
   352→   352→  ctx.fill();
   353→   353→
   354→   354→  ctx.shadowBlur = 0;
   355→   355→  ctx.restore();
   356→   356→}

   359→   359→// Star collectible rendering — 5-pointed star shape
   360→   360→// ---------------------------------------------------------------------------
   361→   361→
   362→   362→/**
   363→   363→ * Draws a proper 5-pointed star path. Generic helper used by drawStarCollectible.
   364→   364→ *
   365→   365→ * @param ctx     Canvas context
   366→   366→ * @param cx      Center X
   367→   367→ * @param cy      Center Y
   368→   368→ * @param outerR  Outer radius (tip of star)
   369→   369→ * @param innerR  Inner radius (valley between points)
   370→   370→ * @param points  Number of points (always 5 for stars)
   371→   371→ * @param rotation Rotation in radians (for spin animation)
   372→   372→ */
   373→   373→export function drawStarShape(
   374→   374→  ctx: CanvasRenderingContext2D,
   375→   375→  cx: number,
   376→   376→  cy: number,
   377→   377→  outerR: number,
   378→   378→  innerR: number,
   379→   379→  points: number,
   380→   380→  rotation: number,
   381→   381→): void {
   382→   382→  ctx.beginPath();
   383→   383→  for (let i = 0; i < points * 2; i++) {
   384→   384→    const r = i % 2 === 0 ? outerR : innerR;
   385→   385→    const angle = (Math.PI / points) * i - Math.PI / 2 + rotation;
   386→   386→    const px = cx + Math.cos(angle) * r;
   387→   387→    const py = cy + Math.sin(angle) * r;
   388→   388→    if (i === 0) ctx.moveTo(px, py);
   389→   389→    else ctx.lineTo(px, py);
   390→   390→  }
   391→   391→  ctx.closePath();
   392→   392→}

   394→   394→/**
   395→   395→ * Draws a 5-pointed star collectible with golden glow, subtle rotation,
   396→   396→ * and pulsing glow effect. Stars are the rare high-value pickups.
   397→   397→ */
   398→   398→export function drawStarCollectible(
   399→   399→  ctx: CanvasRenderingContext2D,
   400→   400→  x: number,
   401→   401→  y: number,
   402→   402→  size: number,
   403→   403→  tick: number,
   404→   404→  lowQuality: boolean,
   405→   405→): void {
   406→   406→  const outerR = Math.max(4, size);
   407→   407→  const innerR = Math.max(2, size * 0.4);
   408→   408→
   409→   409→  // Rotation: slow spin based on tick
   410→   410→  const rotation = tick * 0.002;
   411→   411→
   412→   412→  // Pulsing glow: size oscillates ±1.5px over ~1.5s
   413→   413→  const pulse = lowQuality ? 0 : Math.sin(tick * 0.004) * 1.5;
   414→   414→  const animOuter = outerR + pulse;
   415→   415→  const animInner = innerR + pulse * 0.4;
   416→   416→
   417→   417→  ctx.save();
   418→   418→
   419→   419→  // Glow
   420→   420→  if (!lowQuality) {
   421→   421→    ctx.shadowColor = '#f59e0b';
   422→   422→    ctx.shadowBlur = 12 + Math.sin(tick * 0.003) * 4;
   423→   423→  }
   424→   424→
   425→   425→  // Star shape
   426→   426→  drawStarShape(ctx, x, y, animOuter, animInner, 5, rotation);
   427→   427→
   428→   428→  // Golden radial gradient fill
   429→   429→  const grad = ctx.createRadialGradient(x, y, 0, x, y, animOuter);
   430→   430→  grad.addColorStop(0, '#fef3c7'); // bright center
   431→   431→  grad.addColorStop(0.4, '#fbbf24');
   432→   432→  grad.addColorStop(1, '#f59e0b');
   433→   433→  ctx.fillStyle = grad;
   434→   434→  ctx.fill();
   435→   435→
   436→   436→  ctx.shadowBlur = 0;
   437→   437→  ctx.restore();
   438→   438→}

   440→   440→// ---------------------------------------------------------------------------
   441→   441→// Extraction ring — white-to-green circle near an extracting snake's head
   442→   442→// ---------------------------------------------------------------------------
   443→   443→
   444→   444→/**
   445→   445→ * Draws the extraction progress ring around a snake's head.
   446→   446→ * Visible to ALL players as a warning that someone is banking chips.
   447→   447→ */
   448→   448→export function drawExtractionRing(
   449→   449→  ctx: CanvasRenderingContext2D,
   450→   450→  x: number,
   451→   451→  y: number,
   452→   452→  snakeSize: number,
   453→   453→  progress: number, // 0..1
   454→   454→  zoom: number,
   455→   455→): void {
   456→   456→  if (progress <= 0 || progress > 1) return;
   457→   457→
   458→   458→  const ringRadius = Math.max(8, snakeSize + 10 / zoom);
   459→   459→  const lineWidth = Math.max(2, 3 / zoom);
   460→   460→
   461→   461→  ctx.save();
   462→   462→
   463→   463→  // Background ring (track)
   464→   464→  ctx.beginPath();
   465→   465→  ctx.arc(x, y, ringRadius, 0, Math.PI * 2);
   466→   466→  ctx.strokeStyle = 'rgba(255, 255, 255, 0.15)';
   467→   467→  ctx.lineWidth = lineWidth;
   468→   468→  ctx.stroke();
   469→   469→
   470→   470→  // Progress arc (white → green gradient)
   471→   471→  const startAngle = -Math.PI / 2;
   472→   472→  const endAngle = startAngle + progress * Math.PI * 2;
   473→   473→  ctx.beginPath();
   474→   474→  ctx.arc(x, y, ringRadius, startAngle, endAngle);
   475→   475→
   476→   476→  // Color: white at start → emerald green at end
   477→   477→  const r = Math.round(255 - progress * 227); // 255 → 28
   478→   478→  const g = Math.round(255 - progress * 55);  // 255 → 200
   479→   479→  const b = Math.round(255 - progress * 215); // 255 → 40
   480→   480→  ctx.strokeStyle = `rgba(${r}, ${g}, ${b}, 0.9)`;
   481→   481→  ctx.lineWidth = lineWidth + 1 / zoom;
   482→   482→  ctx.lineCap = 'round';
   483→   483→  ctx.stroke();
   484→   484→
   485→   485→  // Glow
   486→   486→  if (!lowQualityCheck(ctx)) {
   487→   487→    ctx.shadowColor = `rgba(${r}, ${g}, ${b}, 0.5)`;
   488→   488→    ctx.shadowBlur = 8 / zoom;
   489→   489→    ctx.beginPath();
   490→   490→    ctx.arc(x, y, ringRadius, startAngle, endAngle);
   491→   491→    ctx.strokeStyle = `rgba(${r}, ${g}, ${b}, 0.4)`;
   492→   492→    ctx.lineWidth = lineWidth;
   493→   493→    ctx.stroke();
   494→   494→    ctx.shadowBlur = 0;
   495→   495→  }
   496→   496→
   497→   497→  ctx.restore();
   498→   498→}

   500→   500→/** Cheap low-quality check — if canvas context has a hint. */
   501→   501→function lowQualityCheck(_ctx: CanvasRenderingContext2D): boolean {
   502→   502→  return false; // We don't pass lowQuality here; caller can gate
   503→   503→}

   505→   505→// ---------------------------------------------------------------------------
   506→   506→// Food (main entry point — batches regular orbs, draws stars individually)
   507→   507→// ---------------------------------------------------------------------------
   508→   508→
   509→   509→/**
   510→   510→ * Draws all food in two passes:
   511→   511→ *  1. Regular food orbs — batched by orbSize tier into Path2D groups.
   512→   512→ *     Each tier (small/medium/large) has its own glow gradient.
   513→   513→ *  2. Star collectibles — drawn individually as 5-pointed gold stars with
   514→   514→ *     rotation + pulsing glow.
   515→   515→ */
   516→   516→export function drawFood(rc: FrameRenderCtx, foods: FoodSnapshot[]): void {
   517→   517→  const { ctx, lowQuality } = rc;
   518→   518→  const vis = computeVisibleRect(rc);
   519→   519→
   520→   520→  // --- Regular food orbs: batch by value tier ---
   521→   521→  const smallOrbs: FoodSnapshot[] = [];
   522→   522→  const mediumOrbs: FoodSnapshot[] = [];
   523→   523→  const largeOrbs: FoodSnapshot[] = [];
   524→   524→  const starChips: FoodSnapshot[] = [];
   525→   525→
   526→   526→  for (let i = 0; i < foods.length; i++) {
   527→   527→    const f = foods[i];
   528→   528→    if (f.x < vis.left || f.x > vis.right || f.y < vis.top || f.y > vis.bottom) {
   529→   529→      continue; // culled
   530→   530→    }
   531→   531→    if (f.isStarChip) {
   532→   532→      starChips.push(f);
   533→   533→      continue;
   534→   534→    }
   535→   535→    // Route to orb tier by value
   536→   536→    if (f.value >= 5) largeOrbs.push(f);
   537→   537→    else if (f.value >= 3) mediumOrbs.push(f);
   538→   538→    else smallOrbs.push(f);
   539→   539→  }
   540→   540→
   541→   541→  // Draw each orb tier
   542→   542→  const drawOrbBatch = (orbs: FoodSnapshot[]) => {
   543→   543→    for (let i = 0; i < orbs.length; i++) {
   544→   544→      const f = orbs[i];
   545→   545→      drawFoodOrb(ctx, f.x, f.y, f.orbSize ? f.size : f.size, f.value, f.color, f.glowColor ?? '', rc.now, lowQuality);
   546→   546→    }
   547→   547→  };
   548→   548→
   549→   549→  drawOrbBatch(smallOrbs);
   550→   550→  drawOrbBatch(mediumOrbs);
   551→   551→  drawOrbBatch(largeOrbs);
   552→   552→
   553→   553→  // --- Star collectibles: 5-pointed gold stars with rotation + glow + value label ---
   554→   554→  if (starChips.length > 0) {
   555→   555→    ctx.save();
   556→   556→    for (let i = 0; i < starChips.length; i++) {
   557→   557→      const f = starChips[i];
   558→   558→      drawStarCollectible(ctx, f.x, f.y, Math.max(6, f.size + 4), rc.now, lowQuality);
   559→   559→      // Draw chip value label inside the star
   560→   560→      if (f.value > 0) {
   561→   561→        const starRadius = Math.max(4, f.size + 4);
   562→   562→        const labelSize = Math.max(7, Math.min(11, starRadius * 0.55));
   563→   563→        ctx.font = `bold ${labelSize}px monospace`;
   564→   564→        ctx.textAlign = 'center';
   565→   565→        ctx.textBaseline = 'middle';
   566→   566→        ctx.fillStyle = '#7c2d12'; // dark brown, readable inside golden star
   567→   567→        const label = f.value >= 1000 ? `${Math.round(f.value / 1000)}k` : `${Math.round(f.value)}`;
   568→   568→        ctx.fillText(label, f.x, f.y + 0.5);
   569→   569→      }
   570→   570→    }
   571→   571→    ctx.restore();
   572→   572→  }
   573→   573→}

   575→   575→// ---------------------------------------------------------------------------
   576→   576→// Snake rendering
   577→   577→// ---------------------------------------------------------------------------
   578→   578→
   579→   579→/**
   580→   580→ * Draws a single snake. The body is rendered as ONE thick stroked polyline
   581→   581→ * (with an outline underlay). The head is drawn separately with eyes and
   582→   582→ * (for the player's own snake, high-quality only) a glow halo.
   583→   583→ *
   584→   584→ * Player skin patterns (rainbow / neon / metallic / camo) are applied when
   585→   585→ * `rc.playerSkin.pattern` is set and the snake is the local player.
   586→   586→ *
   587→   587→ * @param opacity  Optional global alpha override (0..1). Default = 1.
   588→   588→ */
   589→   589→export function drawSnake(rc: FrameRenderCtx, snake: SnakeSnapshot, opacity?: number): void {
   590→   590→  const pts = snake.points;
   591→   591→  if (!pts || pts.length === 0) return;
   592→   592→
   593→   593→  const vis = computeVisibleRect(rc);
   594→   594→  if (!snakeIsVisible(pts, vis)) return;
   595→   595→
   596→   596→  const { ctx, zoom, lowQuality } = rc;
   597→   597→  const isMe = snake.id === rc.myId;
   598→   598→  const baseAlpha = opacity ?? 1;
   599→   599→
   600→   600→  // Downsample long snakes
   601→   601→  const stride = pts.length > 60 ? 2 : 1;
   602→   602→
   603→   603→  // Body width in world units
   604→   604→  const radius = Math.max(2, snake.visualRadius ?? snake.size);
   605→   605→  const width = radius * 2;
   606→   606→
   607→   607→  ctx.save();
   608→   608→  ctx.globalAlpha = baseAlpha;
   609→   609→
   610→   610→  // --- Outline (underlay) ---
   611→   611→  if (snake.secondaryColor) {
   612→   612→    ctx.save();
   613→   613→    ctx.lineCap = 'round';
   614→   614→    ctx.lineJoin = 'round';
   615→   615→    ctx.lineWidth = width + 4 / zoom;
   616→   616→    ctx.strokeStyle = snake.secondaryColor;
   617→   617→    ctx.globalAlpha = baseAlpha * 0.55;
   618→   618→    ctx.beginPath();
   619→   619→    ctx.moveTo(pts[0].x, pts[0].y);
   620→   620→    for (let i = stride; i < pts.length; i += stride) {
   621→   621→      ctx.lineTo(pts[i].x, pts[i].y);
   622→   622→    }
   623→   623→    ctx.stroke();
   624→   624→    ctx.restore();
   625→   625→  }
   626→   626→
   627→   627→  // --- Body ---
   628→   628→  ctx.save();
   629→   629→  ctx.globalAlpha = baseAlpha;
   630→   630→  ctx.lineCap = 'round';
   631→   631→  ctx.lineJoin = 'round';
   632→   632→  ctx.lineWidth = width;
   633→   633→
   634→   634→  const pattern = isMe ? rc.playerSkin?.pattern : undefined;
   635→   635→  if (pattern === 'metallic' && snake.secondaryColor) {
   636→   636→    const g = getMetallicGradient(rc, radius, snake.color, snake.secondaryColor);
   637→   637→    ctx.strokeStyle = g;
   638→   638→    ctx.beginPath();
   639→   639→    ctx.moveTo(pts[0].x, pts[0].y);
   640→   640→    for (let i = stride; i < pts.length; i += stride) {
   641→   641→      ctx.lineTo(pts[i].x, pts[i].y);
   642→   642→    }
   643→   643→    ctx.stroke();
   644→   644→  } else if (pattern === 'rainbow' && !lowQuality) {
   645→   645→    const chunkSize = Math.max(4, Math.floor(pts.length / 12));
   646→   646→    ctx.lineWidth = width;
   647→   647→    for (let i = 0; i < pts.length - 1; i += chunkSize) {
   648→   648→      const hue = (rc.now * 0.05 + i * 14) % 360;
   649→   649→      ctx.strokeStyle = `hsl(${hue}, 90%, 55%)`;
   650→   650→      ctx.beginPath();
   651→   651→      ctx.moveTo(pts[i].x, pts[i].y);
   652→   652→      const end = Math.min(pts.length - 1, i + chunkSize + 1);
   653→   653→      for (let j = i + stride; j <= end; j += stride) {
   654→   654→        ctx.lineTo(pts[j].x, pts[j].y);
   655→   655→      }
   656→   656→      ctx.stroke();
   657→   657→    }
   658→   658→  } else if (pattern === 'neon' && !lowQuality) {
   659→   659→    const chunkSize = Math.max(4, Math.floor(pts.length / 10));
   660→   660→    ctx.lineWidth = width;
   661→   661→    let chunkIndex = 0;
   662→   662→    for (let i = 0; i < pts.length - 1; i += chunkSize) {
   663→   663→      const ratio = (Math.sin(rc.now * 0.009 - chunkIndex * 0.28) + 1) / 2;
   664→   664→      ctx.strokeStyle = ratio > 0.5 ? '#06b6d4' : '#a855f7';
   665→   665→      ctx.beginPath();
   666→   666→      ctx.moveTo(pts[i].x, pts[i].y);
   667→   667→      const end = Math.min(pts.length - 1, i + chunkSize + 1);
   668→   668→      for (let j = i + stride; j <= end; j += stride) {
   669→   669→        ctx.lineTo(pts[j].x, pts[j].y);
   670→   670→      }
   671→   671→      ctx.stroke();
   672→   672→      chunkIndex += 1;
   673→   673→    }
   674→   674→  } else if (pattern === 'camo') {
   675→   675→    const camoColors = ['#15803d', '#854d0e', '#78350f', '#166534'];
   676→   676→    const chunkSize = Math.max(4, Math.floor(pts.length / 12));
   677→   677→    let chunkIndex = 0;
   678→   678→    for (let i = 0; i < pts.length - 1; i += chunkSize) {
   679→   679→      ctx.strokeStyle = camoColors[chunkIndex % camoColors.length];
   680→   680→      ctx.beginPath();
   681→   681→      ctx.moveTo(pts[i].x, pts[i].y);
   682→   682→      const end = Math.min(pts.length - 1, i + chunkSize + 1);
   683→   683→      for (let j = i + stride; j <= end; j += stride) {
   684→   684→        ctx.lineTo(pts[j].x, pts[j].y);
   685→   685→      }
   686→   686→      ctx.stroke();
   687→   687→      chunkIndex += 1;
   688→   688→    }
   689→   689→  } else {
   690→   690→    // Default: solid snake.color
   691→   691→    ctx.strokeStyle = snake.color;
   692→   692→    ctx.beginPath();
   693→   693→    ctx.moveTo(pts[0].x, pts[0].y);
   694→   694→    for (let i = stride; i < pts.length; i += stride) {
   695→   695→      ctx.lineTo(pts[i].x, pts[i].y);
   696→   696→    }
   697→   697→    ctx.stroke();
   698→   698→  }
   699→   699→  ctx.restore();
   700→   700→
   701→   701→  // --- Head ---
   702→   702→  const head = pts[0];
   703→   703→  ctx.save();
   704→   704→  ctx.globalAlpha = baseAlpha;
   705→   705→  if (isMe && !lowQuality) {
   706→   706→    ctx.shadowColor = snake.color;
   707→   707→    ctx.shadowBlur = 14;
   708→   708→  }
   709→   709→  ctx.beginPath();
   710→   710→  ctx.arc(head.x, head.y, radius, 0, Math.PI * 2);
   711→   711→  ctx.fillStyle = snake.color;
   712→   712→  ctx.fill();
   713→   713→  if (isMe && !lowQuality) ctx.shadowBlur = 0;
   714→   714→
   715→   715→  // Eyes
   716→   716→  const eyeOffset = radius * 0.45;
   717→   717→  const eyeR = Math.max(1.5, radius * 0.32);
   718→   718→  const pupilR = Math.max(0.8, radius * 0.18);
   719→   719→  const angle = snake.angle;
   720→   720→  const perp = angle + Math.PI / 2;
   721→   721→  const ex1 = head.x + Math.cos(angle) * eyeOffset * 0.4 + Math.cos(perp) * eyeOffset;
   722→   722→  const ey1 = head.y + Math.sin(angle) * eyeOffset * 0.4 + Math.sin(perp) * eyeOffset;
   723→   723→  const ex2 = head.x + Math.cos(angle) * eyeOffset * 0.4 - Math.cos(perp) * eyeOffset;
   724→   724→  const ey2 = head.y + Math.sin(angle) * eyeOffset * 0.4 - Math.sin(perp) * eyeOffset;
   725→   725→
   726→   726→  ctx.fillStyle = '#ffffff';
   727→   727→  ctx.beginPath();
   728→   728→  ctx.arc(ex1, ey1, eyeR, 0, Math.PI * 2);
   729→   729→  ctx.arc(ex2, ey2, eyeR, 0, Math.PI * 2);
   730→   730→  ctx.fill();
   731→   731→
   732→   732→  ctx.fillStyle = '#0a0a0a';
   733→   733→  ctx.beginPath();
   734→   734→  ctx.arc(ex1 + Math.cos(angle) * pupilR, ey1 + Math.sin(angle) * pupilR, pupilR, 0, Math.PI * 2);
   735→   735→  ctx.arc(ex2 + Math.cos(angle) * pupilR, ey2 + Math.sin(angle) * pupilR, pupilR, 0, Math.PI * 2);
   736→   736→  ctx.fill();
   737→   737→
   738→   738→  // Spawn-protection ring
   739→   739→  if (snake.spawnProtected) {
   740→   740→    ctx.strokeStyle = 'rgba(255,255,255,0.6)';
   741→   741→    ctx.lineWidth = 2 / zoom;
   742→   742→    ctx.beginPath();
   743→   743→    ctx.arc(head.x, head.y, radius + 4 / zoom, 0, Math.PI * 2);
   744→   744→    ctx.stroke();
   745→   745→  }
   746→   746→
   747→   747→  ctx.restore();
   748→   748→
   749→   749→  // --- Name label ---
   750→   750→  if (snake.name) {
   751→   751→    ctx.save();
   752→   752→    ctx.globalAlpha = baseAlpha;
   753→   753→    ctx.font = `${Math.max(10, 12 / zoom)}px monospace`;
   754→   754→    ctx.textAlign = 'center';
   755→   755→    ctx.textBaseline = 'bottom';
   756→   756→    const labelY = head.y - radius - 6 / zoom;
   757→   757→    // Bots get a distinct muted orange label with [BOT] tag; players get normal label
   758→   758→    if (snake.isBot) {
   759→   759→      ctx.fillStyle = 'rgba(251, 146, 60, 0.75)'; // orange-400 muted
   760→   760→      const botLabel = `[BOT] ${snake.name}`;
   761→   761→      ctx.fillText(botLabel, head.x, labelY);
   762→   762→    } else {
   763→   763→      ctx.fillStyle = isMe ? '#22c55e' : 'rgba(226, 232, 240, 0.85)';
   764→   764→      ctx.fillText(snake.name, head.x, labelY);
   765→   765→    }
   766→   766→    if (snake.userTag) {
   767→   767→      ctx.fillStyle = 'rgba(148, 163, 184, 0.7)';
   768→   768→      ctx.font = `${Math.max(8, 9 / zoom)}px monospace`;
   769→   769→      ctx.fillText(snake.userTag, head.x, labelY - 12 / zoom);
   770→   770→    }
   771→   771→    ctx.restore();
   772→   772→  }
   773→   773→
   774→   774→  // --- Chat bubble ---
   775→   775→  if (snake.chatMessage) {
   776→   776→    ctx.save();
   777→   777→    ctx.globalAlpha = baseAlpha;
   778→   778→    drawChatBubble(rc, head.x, head.y - radius - 24 / zoom, snake.chatMessage);
   779→   779→    ctx.restore();
   780→   780→  }
   781→   781→
   782→   782→  ctx.restore(); // outer globalAlpha save
   783→   783→}
   784→   784→
   785→   785→// ---------------------------------------------------------------------------
   786→   786→// Snake opacity layering system
   787→   787→// ---------------------------------------------------------------------------
   788→   788→
   789→   789→/** Proximity threshold for opacity fade (world-space pixels). */
   790→   790→const LAYERING_PROXIMITY = 30;
   791→   791→
   792→   792→/**
   793→   793→ * Draws a snake with opacity layering: when a smaller snake passes close to a
   794→   794→ * larger snake, the larger snake fades to 75% opacity. This creates visual
   795→   795→ * depth when snakes overlap spatially.
   796→   796→ *
   797→   797→ * Logic:
   798→   798→ *  1. For each OTHER snake, check if its head is within ~30px of this snake's head
   799→   799→ *  2. If a smaller snake (by size) is nearby, this snake fades to 0.75 opacity
   800→   800→ *  3. Otherwise, render at full opacity
   801→   801→ */
   802→   802→export function drawSnakeWithLayering(
   803→   803→  rc: FrameRenderCtx,
   804→   804→  snake: SnakeSnapshot,
   805→   805→  allSnakes: SnakeSnapshot[],
   806→   806→): void {
   807→   807→  const pts = snake.points;
   808→   808→  if (!pts || pts.length === 0) return;
   809→   809→
   810→   810→  const myHead = pts[0];
   811→   811→  let shouldFade = false;
   812→   812→
   813→   813→  for (let i = 0; i < allSnakes.length; i++) {
   814→   814→    const other = allSnakes[i];
   815→   815→    if (other.id === snake.id) continue;
   816→   816→    const otherPts = other.points;
   817→   817→    if (!otherPts || otherPts.length === 0) continue;
   818→   818→
   819→   819→    const otherHead = otherPts[0];
   820→   820→    const dist = Math.hypot(otherHead.x - myHead.x, otherHead.y - myHead.y);
   821→   821→
   822→   822→    const snakeR = snake.visualRadius ?? snake.size;
   823→   823→    const otherR = other.visualRadius ?? other.size;
   824→   824→    if (dist < LAYERING_PROXIMITY && snakeR > otherR) {
   825→   825→      // A smaller snake is close by — fade this (larger) snake
   826→   826→      shouldFade = true;
   827→   827→      break;
   828→   828→    }
   829→   829→  }
   830→   830→
   831→   831→  drawSnake(rc, snake, shouldFade ? 0.75 : 1);
   832→   832→}
   833→   833→
   834→   834→// ---------------------------------------------------------------------------
   835→   835→// Chip label above snake head
   836→   836→// ---------------------------------------------------------------------------
   837→   837→
   838→   838→/**
   839→   839→ * Draws a chip count label above the snake's head. Uses Indian numbering
   840→   840→ * (K, L, Cr). Only meaningful for real players (bots have 0 carriedChips).
   841→   841→ *
   842→   842→ * Rendered as a semi-transparent pill with white text, positioned centered
   843→   843→ * above the head.
   844→   844→ */
   845→   845→export function drawChipLabel(
   846→   846→  ctx: CanvasRenderingContext2D,
   847→   847→  x: number,
   848→   848→  y: number,
   849→   849→  chips: number,
   850→   850→  snakeSize: number,
   851→   851→  zoom: number,
   852→   852→): void {
   853→   853→  if (chips <= 0) return; // bots have 0 chips, skip them
   854→   854→
   855→   855→  const label = formatChipDisplay(chips);
   856→   856→  const fontSize = Math.max(9, 10 / zoom);
   857→   857→  const offset = snakeSize + 12 / zoom;
   858→   858→
   859→   859→  ctx.save();
   860→   860→  ctx.font = `bold ${fontSize}px monospace`;
   861→   861→  ctx.textAlign = 'center';
   862→   862→  ctx.textBaseline = 'bottom';
   863→   863→
   864→   864→  const metrics = ctx.measureText(label);
   865→   865→  const padX = 5 / zoom;
   866→   866→  const padY = 2 / zoom;
   867→   867→  const tw = metrics.width + padX * 2;
   868→   868→  const th = fontSize + padY * 2;
   869→   869→  const rx = x - tw / 2;
   870→   870→  const ry = y - offset - th;
   871→   871→  const rr = 4 / zoom;
   872→   872→
   873→   873→  // Background pill
   874→   874→  ctx.fillStyle = 'rgba(0, 0, 0, 0.65)';
   875→   875→  ctx.beginPath();
   876→   876→  ctx.moveTo(rx + rr, ry);
   877→   877→  ctx.lineTo(rx + tw - rr, ry);
   878→   878→  ctx.arcTo(rx + tw, ry, rx + tw, ry + rr, rr);
   879→   879→  ctx.lineTo(rx + tw, ry + th - rr);
   880→   880→  ctx.arcTo(rx + tw, ry + th, rx + tw - rr, ry + th, rr);
   881→   881→  ctx.lineTo(rx + rr, ry + th);
   882→   882→  ctx.arcTo(rx, ry + th, rx, ry + th - rr, rr);
   883→   883→  ctx.lineTo(rx, ry + rr);
   884→   884→  ctx.arcTo(rx, ry, rx + rr, ry, rr);
   885→   885→  ctx.closePath();
   886→   886→  ctx.fill();
   887→   887→
   888→   888→  // Gold accent border
   889→   889→  ctx.strokeStyle = 'rgba(251, 191, 36, 0.7)';
   890→   890→  ctx.lineWidth = 1 / zoom;
   891→   891→  ctx.stroke();
   892→   892→
   893→   893→  // Text
   894→   894→  ctx.fillStyle = '#ffffff';
   895→   895→  ctx.textBaseline = 'middle';
   896→   896→  ctx.fillText(label, x, ry + th / 2);
   897→   897→
   898→   898→  ctx.restore();
   899→   899→}
   900→   900→
   901→   901→// ---------------------------------------------------------------------------
   902→   902→// Chat bubble
   903→   903→// ---------------------------------------------------------------------------
   904→   904→
   905→   905→/** Draws a small chat bubble above a snake head. */
   906→   906→function drawChatBubble(
   907→   907→  rc: FrameRenderCtx,
   908→   908→  x: number,
   909→   909→  y: number,
   910→   910→  msg: string,
   911→   911→): void {
   912→   912→  const { ctx, zoom } = rc;
   913→   913→  const text = msg.length > 80 ? msg.slice(0, 78) + '…' : msg;
   914→   914→  ctx.save();
   915→   915→  ctx.font = `${Math.max(10, 11 / zoom)}px monospace`;
   916→   916→  const metrics = ctx.measureText(text);
   917→   917→  const padX = 6 / zoom;
   918→   918→  const padY = 3 / zoom;
   919→   919→  const tw = metrics.width + padX * 2;
   920→   920→  const th = 14 / zoom + padY * 2;
   921→   921→  ctx.fillStyle = 'rgba(15, 23, 42, 0.92)';
   922→   922→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.6)';
   923→   923→  ctx.lineWidth = 1.5 / zoom;
   924→   924→  const rx = x - tw / 2;
   925→   925→  const ry = y - th;
   926→   926→  const rr = 4 / zoom;
   927→   927→  ctx.beginPath();
   928→   928→  ctx.moveTo(rx + rr, ry);
   929→   929→  ctx.lineTo(rx + tw - rr, ry);
   930→   930→  ctx.arcTo(rx + tw, ry, rx + tw, ry + rr, rr);
   931→   931→  ctx.lineTo(rx + tw, ry + th - rr);
   932→   932→  ctx.arcTo(rx + tw, ry + th, rx + tw - rr, ry + th, rr);
   933→   933→  ctx.lineTo(rx + rr, ry + th);
   934→   934→  ctx.arcTo(rx, ry + th, rx, ry + th - rr, rr);
   935→   935→  ctx.lineTo(rx, ry + rr);
   936→   936→  ctx.arcTo(rx, ry, rx + rr, ry, rr);
   937→   937→  ctx.closePath();
   938→   938→  ctx.fill();
   939→   939→  ctx.stroke();
   940→   940→  ctx.fillStyle = '#f8fafc';
   941→   941→  ctx.textAlign = 'center';
   942→   942→  ctx.textBaseline = 'middle';
   943→   943→  ctx.fillText(text, x, y - th / 2);
   944→   944→  ctx.restore();
   945→   945→}

   951→   951→/** Draws the particle array (caller caps at MAX_PARTICLES = 200). */
   952→   952→export function drawParticles(rc: FrameRenderCtx, particles: Particle[]): void {
   953→   953→  if (particles.length === 0) return;
   954→   954→  const { ctx, zoom } = rc;
   955→   955→  ctx.save();
   956→   956→  ctx.globalCompositeOperation = 'lighter';
   957→   957→  for (let i = 0; i < particles.length; i++) {
   958→   958→    const p = particles[i];
   959→   959→    if (p.life <= 0) continue;
   960→   960→    const alpha = Math.max(0, Math.min(1, p.life / p.maxLife));
   961→   961→    ctx.globalAlpha = alpha;
   962→   962→    ctx.fillStyle = p.color;
   963→   963→    ctx.beginPath();
   964→   964→    ctx.arc(p.x, p.y, p.size / zoom, 0, Math.PI * 2);
   965→   965→    ctx.fill();
   966→   966→  }
   967→   967→  ctx.globalAlpha = 1;
   968→   968→  ctx.globalCompositeOperation = 'source-over';
   969→   969→  ctx.restore();
   970→   970→}

   973→   973→// Minimap (drawn in screen-space, no camera transform)
   974→   974→// ---------------------------------------------------------------------------
   975→   975→
   976→   976→export interface MinimapArgs {
   977→   977→  ctx: CanvasRenderingContext2D;
   978→   978→  /** Top-left of minimap in CSS pixels. */
   979→   979→  x: number;
   980→   980→  y: number;
   981→   981→  /** Minimap size (square). */
   982→   982→  size: number;
   983→   983→  worldSize: number;
   984→   984→  /** Current breathing arena radius (world units). */
   985→   985→  arenaRadius: number;
   986→   986→  snakes: SnakeSnapshot[];
   987→   987→  myId: string;
   988→   988→  /**
   989→   989→   * World-space radius around the player the radar should cover. Snakes
   990→   990→   * outside this range are NOT rendered on the small corner minimap.
   991→   991→   * Defaults to WORLD_SIZE/2 (legacy full-radar behaviour) when omitted.
   992→   992→   */
   993→   993→  range?: number;
   994→   994→}
   995→   995→
   996→   996→/**
   997→   997→ * Draws the circular radar minimap. Player is rendered as an indigo dot at
   998→   998→ * the centre (the radar is centered on the player's head). The arena
   999→   999→ * boundary is a dashed rose circle. Bots render as rose dots, other real
  1000→  1000→ * players as emerald dots — matches AUDIT-A radar visual language.
  1001→  1001→ */
  1002→  1002→export function drawMinimap(args: MinimapArgs): void {
  1003→  1003→  const { ctx, x, y, size, worldSize, arenaRadius, snakes, myId } = args;
  1004→  1004→  const cx = x + size / 2;
  1005→  1005→  const cy = y + size / 2;
  1006→  1006→  const r = size / 2;
  1007→  1007→  const radarRange = args.range ?? WORLD_SIZE / 2;
  1008→  1008→  const scale = r / radarRange;
  1009→  1009→
  1010→  1010→  ctx.save();
  1011→  1011→  // Background circle (radar disc)
  1012→  1012→  ctx.beginPath();
  1013→  1013→  ctx.arc(cx, cy, r, 0, Math.PI * 2);
  1014→  1014→  ctx.fillStyle = 'rgba(2, 6, 23, 0.85)';
  1015→  1015→  ctx.fill();
  1016→  1016→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.4)';
  1017→  1017→  ctx.lineWidth = 1.5;
  1018→  1018→  ctx.stroke();
  1019→  1019→
  1020→  1020→  // Clip to minimap circle.
  1021→  1021→  ctx.beginPath();
  1022→  1022→  ctx.arc(cx, cy, r, 0, Math.PI * 2);
  1023→  1023→  ctx.clip();
  1024→  1024→
  1025→  1025→  // Concentric rings.
  1026→  1026→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.10)';
  1027→  1027→  ctx.lineWidth = 1;
  1028→  1028→  for (const inset of [2, 5, 8]) {
  1029→  1029→    ctx.beginPath();
  1030→  1030→    ctx.arc(cx, cy, r - inset, 0, Math.PI * 2);
  1031→  1031→    ctx.stroke();
  1032→  1032→  }
  1033→  1033→
  1034→  1034→  // Crosshairs.
  1035→  1035→  ctx.strokeStyle = 'rgba(15, 23, 42, 0.4)';
  1036→  1036→  ctx.beginPath();
  1037→  1037→  ctx.moveTo(cx, y);
  1038→  1038→  ctx.lineTo(cx, y + size);
  1039→  1039→  ctx.moveTo(x, cy);
  1040→  1040→  ctx.lineTo(x + size, cy);
  1041→  1041→  ctx.stroke();
  1042→  1042→
  1043→  1043→  // Arena boundary (dashed rose) — centered on world center.
  1044→  1044→  const mySnake = snakes.find((s) => s.id === myId);
  1045→  1045→  const myHead = mySnake?.points?.[0];
  1046→  1046→  const px = myHead ? myHead.x : worldSize / 2;
  1047→  1047→  const py = myHead ? myHead.y : worldSize / 2;
  1048→  1048→  const worldCenterOffsetX = (worldSize / 2 - px) * scale;
  1049→  1049→  const worldCenterOffsetY = (worldSize / 2 - py) * scale;
  1050→  1050→  const arenaR = arenaRadius * scale;
  1051→  1051→  ctx.strokeStyle = 'rgba(244, 63, 94, 0.6)';
  1052→  1052→  ctx.lineWidth = 1.2;
  1053→  1053→  ctx.setLineDash([4, 4]);
  1054→  1054→  ctx.beginPath();
  1055→  1055→  ctx.arc(cx + worldCenterOffsetX, cy + worldCenterOffsetY, arenaR, 0, Math.PI * 2);
  1056→  1056→  ctx.stroke();
  1057→  1057→  ctx.setLineDash([]);
  1058→  1058→
  1059→  1059→  // Snakes (player centered → drawn at centre; others offset).
  1060→  1060→  for (let i = 0; i < snakes.length; i++) {
  1061→  1061→    const s = snakes[i];
  1062→  1062→    if (!s.points || s.points.length === 0) continue;
  1063→  1063→    const head = s.points[0];
  1064→  1064→    const dx = (head.x - px) * scale;
  1065→  1065→    const dy = (head.y - py) * scale;
  1066→  1066→    const dist = Math.hypot(dx, dy);
  1067→  1067→    if (dist > r) continue;
  1068→  1068→    const mx = cx + dx;
  1069→  1069→    const my = cy + dy;
  1070→  1070→    ctx.beginPath();
  1071→  1071→    if (s.id === myId) {
  1072→  1072→      ctx.arc(mx, my, 3, 0, Math.PI * 2);
  1073→  1073→      ctx.fillStyle = '#818cf8'; // indigo-400
  1074→  1074→    } else if (s.isBot) {
  1075→  1075→      ctx.arc(mx, my, 2, 0, Math.PI * 2);
  1076→  1076→      ctx.fillStyle = '#f43f5e'; // rose-500
  1077→  1077→    } else {
  1078→  1078→      ctx.arc(mx, my, 2, 0, Math.PI * 2);
  1079→  1079→      ctx.fillStyle = '#34d399'; // emerald-400
  1080→  1080→    }
  1081→  1081→    ctx.fill();
  1082→  1082→  }
  1083→  1083→  ctx.restore();
  1084→  1084→}

  1104→  1104→/**
  1105→  1105→ * Draws a full-screen overlay of the entire arena. The arena circle is
  1106→  1106→ * centered on screen and scaled to fit the smaller of (w, h) with padding.
  1107→  1107→ * All snakes are rendered as dots — player = indigo (larger), bots = rose,
  1108→  1108→ * other real humans = emerald. A "Press M to close" hint is drawn at the
  1109→  1109→ * bottom. The player's own dot pulses with a ring.
  1110→  1110→ */
  1111→  1111→export function drawFullMap(args: FullMapArgs): void {
  1112→  1112→  const { ctx, w, h, worldSize, arenaRadius, snakes, myId } = args;
  1113→  1113→  ctx.save();
  1114→  1114→  ctx.setTransform(1, 0, 0, 1, 0, 0);
  1115→  1115→
  1116→  1116→  // --- Background fill (dim slate) ---
  1117→  1117→  ctx.fillStyle = 'rgba(2, 6, 23, 0.94)';
  1118→  1118→  ctx.fillRect(0, 0, w, h);
  1119→  1119→
  1120→  1120→  // --- Layout: arena centred on screen ---
  1121→  1121→  const cx = w / 2;
  1122→  1122→  const cy = h / 2;
  1123→  1123→  const margin = 80;
  1124→  1124→  const fitDim = Math.min(w, h) - margin * 2;
  1125→  1125→  const arenaDiameter = arenaRadius * 2;
  1126→  1126→  const scale = fitDim / arenaDiameter;
  1127→  1127→  const screenR = arenaRadius * scale;
  1128→  1128→
  1129→  1129→  const wcx = worldSize / 2;
  1130→  1130→  const wcy = worldSize / 2;
  1131→  1131→  const toScreenX = (wx: number) => cx + (wx - wcx) * scale;
  1132→  1132→  const toScreenY = (wy: number) => cy + (wy - wcy) * scale;
  1133→  1133→
  1134→  1134→  // --- Concentric range rings (faint) ---
  1135→  1135→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.10)';
  1136→  1136→  ctx.lineWidth = 1;
  1137→  1137→  for (const frac of [0.25, 0.5, 0.75]) {
  1138→  1138→    ctx.beginPath();
  1139→  1139→    ctx.arc(cx, cy, screenR * frac, 0, Math.PI * 2);
  1140→  1140→    ctx.stroke();
  1141→  1141→  }
  1142→  1142→  // Crosshairs.
  1143→  1143→  ctx.strokeStyle = 'rgba(15, 23, 42, 0.5)';
  1144→  1144→  ctx.beginPath();
  1145→  1145→  ctx.moveTo(cx, cy - screenR);
  1146→  1146→  ctx.lineTo(cx, cy + screenR);
  1147→  1147→  ctx.moveTo(cx - screenR, cy);
  1148→  1148→  ctx.lineTo(cx + screenR, cy);
  1149→  1149→  ctx.stroke();
  1150→  1150→
  1151→  1151→  // --- Arena boundary (dashed rose) ---
  1152→  1152→  ctx.strokeStyle = 'rgba(244, 63, 94, 0.8)';
  1153→  1153→  ctx.lineWidth = 2;
  1154→  1154→  ctx.setLineDash([6, 6]);
  1155→  1155→  ctx.beginPath();
  1156→  1156→  ctx.arc(cx, cy, screenR, 0, Math.PI * 2);
  1157→  1157→  ctx.stroke();
  1158→  1158→  ctx.setLineDash([]);
  1159→  1159→
  1160→  1160→  // --- Snakes ---
  1161→  1161→  for (let i = 0; i < snakes.length; i++) {
  1162→  1162→    const s = snakes[i];
  1163→  1163→    if (!s.points || s.points.length === 0) continue;
  1164→  1164→    const head = s.points[0];
  1165→  1165→    const sx = toScreenX(head.x);
  1166→  1166→    const sy = toScreenY(head.y);
  1167→  1167→    ctx.beginPath();
  1168→  1168→    if (s.id === myId) {
  1169→  1169→      ctx.arc(sx, sy, 5, 0, Math.PI * 2);
  1170→  1170→      ctx.fillStyle = '#818cf8';
  1171→  1171→      ctx.fill();
  1172→  1172→      ctx.strokeStyle = 'rgba(129, 140, 248, 0.7)';
  1173→  1173→      ctx.lineWidth = 2;
  1174→  1174→      ctx.beginPath();
  1175→  1175→      ctx.arc(sx, sy, 9, 0, Math.PI * 2);
  1176→  1176→      ctx.stroke();
  1177→  1177→    } else if (s.isBot) {
  1178→  1178→      ctx.arc(sx, sy, 2.5, 0, Math.PI * 2);
  1179→  1179→      ctx.fillStyle = '#f43f5e';
  1180→  1180→      ctx.fill();
  1181→  1181→    } else {
  1182→  1182→      ctx.arc(sx, sy, 3, 0, Math.PI * 2);
  1183→  1183→      ctx.fillStyle = '#34d399';
  1184→  1184→      ctx.fill();
  1185→  1185→    }
  1186→  1186→  }

  1192→  1192→  ctx.textBaseline = 'top';
  1193→  1193→  ctx.fillText('ARENA OVERVIEW — ALL SNAKES', cx, 16);
  1194→  1194→  ctx.fillStyle = 'rgba(148, 163, 184, 0.85)';
  1195→  1195→  ctx.font = '11px monospace';

  1200→  1200→  ctx.font = '11px monospace';
  1201→  1201→  const legendX = 20;

  1204→  1204→    ctx.beginPath();
  1205→  1205→    ctx.arc(legendX + 6, legendY + 6, dotR, 0, Math.PI * 2);

  1212→  1212→  drawLegend('#818cf8', 'You', 5);

  1216→  1216→  ctx.restore();
  1217→  1217→}
