   271→export const MAP_MIN_RADIUS = 3000;  // radius when 1 player (doubled for comfort)
   272→export const MAP_MAX_RADIUS = 16000;  // radius when 1000 players (DOUBLED for 1000-player density)
   273→export const MAP_BREATH_AMPLITUDE = 40;  // breathing oscillation
   274→export const MAP_BREATH_CYCLE_MS = 10000;
   275→export const MAX_ARENA_PLAYERS = 1000;
   276→
   277→/** Compute dynamic map radius based on real player count. */
   278→export function getDynamicMapRadius(realPlayerCount: number, elapsedMs?: number): number {
   279→  const minP = 1;
   280→  const maxP = MAX_ARENA_PLAYERS;
   281→  const count = Math.max(minP, Math.min(maxP, realPlayerCount));
   282→  // sqrt scaling: 1 player -> 1500, ~31 players -> ~3000, 1000 players -> 5000
   283→  const baseRadius = MAP_MIN_RADIUS + (MAP_MAX_RADIUS - MAP_MIN_RADIUS) * Math.sqrt((count - 1) / (maxP - 1));
   284→  // Add breathing
   285→  if (elapsedMs !== undefined) {
   286→    const cycle = (elapsedMs % MAP_BREATH_CYCLE_MS) / MAP_BREATH_CYCLE_MS;
   287→    return baseRadius + Math.sin(cycle * Math.PI * 2) * MAP_BREATH_AMPLITUDE;
   288→  }
   289→  return baseRadius;
   290→}
