   210→// ---------------------------------------------------------------------------
   211→// Local-storage helpers — never crash on parse error
   212→// ---------------------------------------------------------------------------
   213→function readJSON<T>(key: string, fallback: T): T {
   214→  try {
   215→    const raw = localStorage.getItem(key);
   216→    if (!raw) return fallback;
   217→    return JSON.parse(raw) as T;
   218→  } catch {
   219→    return fallback;
   220→  }
   221→}
   222→
   223→function writeJSON(key: string, value: unknown) {
   224→  try {
   225→    localStorage.setItem(key, JSON.stringify(value));
   226→  } catch {
   227→    /* quota errors silently ignored */
   228→  }
   229→}

   231→// Mocked friend balance — original SocialPanel had `getFriendSimulatedChips`.
   232→// We hardcode a friendly number for the co-op modal's eligibility math.
   233→function getFriendSimulatedChips(friend: Friend): number {
   234→  // Stable pseudo-random based on the friend id so each friend has a
   235→  // believable bankroll that doesn't change on re-render.
   236→  let h = 0;
   237→  for (let i = 0; i < friend.id.length; i++) {
   238→    h = (h * 31 + friend.id.charCodeAt(i)) & 0xffffffff;
   239→  }
   240→  return 750 + (Math.abs(h) % 9000);
   241→}

   445→  async function handleSaveProfile() {
   446→    const trimmed = newName.trim();
   447→    if (!trimmed) {
   448→      notify('Nickname cannot be empty!', 'error', onToast);
   449→      return;

   459→        method: 'PUT',
   460→        headers: { 'Content-Type': 'application/json' },
   461→        body: JSON.stringify({
   462→          name: trimmed,
   463→          country: selectedCountry,
   464→          avatar:
   465→            selectedAvatar && selectedAvatar.length <= 8

   467→              : undefined,
   468→          instagram: instagram.trim(),
   469→          youtube: youtube.trim(),
   470→          twitch: twitch.trim(),
   471→        }),
   472→      });
   473→      const data = (await res.json().catch(() => ({}))) as { error?: string };
   474→      if (!res.ok) {
   475→        notify(data?.error || 'Failed to save profile.', 'error', onToast);
   476→        return;
   477→      }

   479→      // Persist socials client-side (server whitelist doesn't include them)

   481→        instagram: instagram.trim(),
   482→        youtube: youtube.trim(),
   483→        twitch: twitch.trim(),
   484→      });

   488→      const countryChanged = selectedCountry !== (player.country || 'US');

   491→          id: 'log-' + Math.random().toString(36).slice(2, 11),

   507→        'Handshake secure! Profile & Social links saved successfully! 🔒',

   512→      notify('Network error. Please try again.', 'error', onToast);

   574→  function handleSendCoOpInvite() {
   575→    if (!activeInviteFriend) return;
   576→    const selectedTier =
   577→      ARENA_TIERS.find((t) => t.id === inviteSelectedArenaId) || ARENA_TIERS[0];
   578→    const friendChips = getFriendSimulatedChips(activeInviteFriend);
   579→    const playerChips = player.bankedChips;
   580→
   581→    if (playerChips < selectedTier.buyIn) {
   582→      notify(
   583→        `You do not have enough chips for ${selectedTier.name}!`,
   584→        'error',
   585→        onToast,
   586→      );
   587→      return;
   588→    }
   589→
   590→    if (friendChips < selectedTier.buyIn) {
   591→      const affordableTiers = ARENA_TIERS.filter(
   592→        (t) => friendChips >= t.buyIn && playerChips >= t.buyIn,
   593→      );
   594→      if (affordableTiers.length > 0) {
   595→        const counterTier = affordableTiers[affordableTiers.length - 1];
   596→        setInviteStatusMessage({
   597→          type: 'counter',
   598→          text: `Sorry! I don't have enough chips for ${selectedTier.name} (need ${selectedTier.buyIn.toLocaleString()} c, only have ${friendChips.toLocaleString()} c). Let's join the "${counterTier.name}" (Buy-In: ${counterTier.buyIn.toLocaleString()} c) instead! Re-invite me?`,
   599→          counterArenaId: counterTier.id,
   600→        });
   601→        notify(
